################ script: zipfrun-intro.py (dossier's zipfrun2.py with port 7101 and a private dir) ################
import socket, random, bisect, collections, sys, time
HOST,PORT="127.0.0.1",7101
class R:
    def __init__(s):
        s.s=socket.create_connection((HOST,PORT)); s.s.setsockopt(socket.IPPROTO_TCP,socket.TCP_NODELAY,1); s.f=s.s.makefile('rb')
    def cmd(s,*a):
        out=b"*%d\r\n"%len(a)
        for x in a:
            x=x if isinstance(x,bytes) else str(x).encode()
            out+=b"$%d\r\n%s\r\n"%(len(x),x)
        s.s.sendall(out); return s.rd()
    def rd(s):
        l=s.f.readline(); t=l[:1]; b=l[1:-2]
        if t in b'+-:': return b if t!=b':' else int(b)
        if t==b'$':
            n=int(b)
            if n<0: return None
            d=s.f.read(n+2); return d[:-2]
        if t==b'*': return [s.rd() for _ in range(int(b))]
def info(r):
    d={}
    for line in r.cmd("INFO").decode().splitlines():
        if ':' in line and not line.startswith('#'):
            k,v=line.split(':',1); d[k]=v
    return d
N=100_000; VAL=b"v"*100
def zipf_seq(s,n,seed):
    rng=random.Random(seed)
    w=[1/(i**s) for i in range(1,N+1)]
    cum=[];c=0
    for x in w: c+=x; cum.append(c)
    return [bisect.bisect_left(cum,rng.random()*c) for _ in range(n)], w
def lru_ratio(seq,C,warm):
    od=collections.OrderedDict(); hit=tot=0
    for i,k in enumerate(seq):
        h=k in od
        if h: od.move_to_end(k)
        else:
            od[k]=1
            if len(od)>C: od.popitem(last=False)
        if i>=warm: tot+=1; hit+=h
    return hit/tot
import subprocess, os
B="/opt/homebrew/opt/redis/bin"
DIR=open("redisdir.txt").read().strip()
def restart():
    global r
    try: subprocess.run([B+"/redis-cli","-p","7101","shutdown","nosave"],capture_output=True)
    except Exception: pass
    time.sleep(0.5)
    subprocess.run([B+"/redis-server","--port","7101","--dir",DIR,"--save","","--daemonize","yes","--logfile",DIR+"/redis.log"],capture_output=True)
    time.sleep(0.7)
    r=R()
def enc(*a):
    out=b"*%d\r\n"%len(a)
    for x in a:
        x=x if isinstance(x,bytes) else str(x).encode()
        out+=b"$%d\r\n%s\r\n"%(len(x),x)
    return out
def batch(r,keys):
    r.s.sendall(b"".join(enc("GET",k) for k in keys))
    miss=[k for k in keys if r.rd() is None]
    if miss:
        r.s.sendall(b"".join(enc("SET",k,VAL) for k in miss))
        for _ in miss: r.rd()
B_=100
def run(seq,lo,hi):
    for i in range(lo,hi,B_):
        batch(r,[f"item:{k:06d}" for k in seq[i:i+B_]])
for s in [1.0,0.8]:
    seq,w=zipf_seq(s,1_500_000,42)
    for frac in [0.01,0.05,0.10,0.20]:
        C=int(N*frac)
        restart()
        r.cmd("CONFIG","SET","maxmemory-policy","allkeys-lru")
        r.cmd("CONFIG","SET","maxmemory-samples","5")
        base0=int(info(r)["used_memory"])
        mm=base0+C*165
        r.cmd("CONFIG","SET","maxmemory",str(mm))
        t=time.time(); pos=0
        for rnd in range(16):                     # tuning rounds: 40,000 requests each
            run(seq,pos,pos+40_000); pos+=40_000
            d=info(r); n=r.cmd("DBSIZE"); um=int(d["used_memory"])
            pk=max((um-base0)/max(n,1),100)
            mm=int(max(base0+C*100, min(mm+(C-n)*pk*0.7, base0+C*400)))
            r.cmd("CONFIG","SET","maxmemory",str(mm))
        run(seq,pos,pos+300_000); pos+=300_000    # settle
        n_before=r.cmd("DBSIZE")
        r.cmd("CONFIG","RESETSTAT")
        run(seq,pos,len(seq))
        d=info(r); h,m=int(d["keyspace_hits"]),int(d["keyspace_misses"])
        ideal=sum(w[:C])/sum(w)
        print(f"s={s} cache={frac*100:>4g}% of items (target {C} keys, held {n_before}, maxmemory={mm} B): Redis allkeys-lru samples=5 hit ratio={h/(h+m)*100:5.1f}%  exact-LRU sim={lru_ratio(seq,C,pos)*100:5.1f}%  ideal static top-C={ideal*100:5.1f}%  measured over {h+m} requests  ({time.time()-t:.0f}s)",flush=True)
restart()

- Breslau, Cao, Fan, Phillips, Shenker, "Web Caching and Zipf-like Distributions: Evidence and Implications", INFOCOM 1999
  (http://www.cs.wisc.edu/~cao/papers/zipf-like.ps.gz, text extracted from the PostScript on 2026-10-27):
  "The value of alpha varies from trace to trace, ranging from 0.64 to 0.83" (Table I: DEC 0.77, UPisa 0.78, FuNet 0.83, UCB 0.69, Questnet 0.73, NLANR 0.64).
  "the top 1% of the documents account for about 20% to 35% of all requests seen by the proxy, and the top 10% of the
  documents account for about 45% to 55% of all requests. However, it takes 25% to 40% of all documents to account for 70%
  of the requests, and it takes 70% to 80% of documents to account for 90% of requests." The "10/90" rule does not apply.
  For alpha = 1, hit ratio H(C) grows as ln C; for 0 < alpha < 1, as C^(1-alpha).
- Bronson, Aghayev, Charapko, Zhu, "Metastable Failures in Distributed Systems", HotOS 2021, section 2.2 "Look-aside Cache"
  (PDF read 2026-10-27): with a 90% hit rate the application handles 3,000 QPS against a database that serves 300;
  if cache contents are lost the database is overloaded, and because the application's timeout treats slow queries as
  failed the cache stays empty: "losing a cache with a 90% hit-rate causes a 10x query amplification."
