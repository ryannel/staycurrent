--- script check4.mjs (exact per-rank sums; 'shared' = one LRU over product+stock pairs, 2,229 bytes a pair)
// independent re-implementation, exact per-rank sums
function shop({s=0.9, reads=1000, sales=20, edits=1, gb=3, ttl=86400, n=2e6, shared=false}){
  let H=0; const w=new Float64Array(n); for(let i=0;i<n;i++){w[i]=1/Math.pow(i+1,s);H+=w[i];}
  const per = shared ? 2229 : 2131; const cap=Math.floor(gb*2**30/per);
  const p=(r,a)=>Math.min(r*ttl/(1+r*ttl), a===Infinity?1:1-Math.exp(-r*a));
  const sums=a=>{let k=0,h=0; for(let i=0;i<n;i++){const q=w[i]/H, r=reads*q, x=p(r,a); k+=x; h+=q*x;} return {k,h};};
  let f=sums(Infinity), lim='ttl';
  if(f.k>cap){let lo=0,hi=1e9; for(let j=0;j<80;j++){const m=(lo+hi)/2; if(sums(m).k<cap) lo=m; else hi=m;} f=sums(lo); lim='memory';}
  // stock: unlimited (writer's model) or shared
  let sh=0; for(let i=0;i<n;i++){const q=w[i]/H, r=reads*q; sh+=q*(shared? p(r, Infinity)*0+0 : r*ttl/(1+r*ttl));}
  let sale=0; for(let i=0;i<n;i++){const q=w[i]/H, r=reads*q, sd=sales*q; sale+=sd*r/(r+sd);}
  const stockHit = shared ? f.h : sh;
  const prod=reads*(1-f.h)+edits*Math.min(1,f.k/n), stock=reads*(1-stockHit)+sale, chk=sales*3;
  return {hit:+f.h.toFixed(4), lim, prod:+prod.toFixed(1), stock:+stock.toFixed(1), chk, total:+(prod+stock+chk).toFixed(1)};
}

================ M4: cold start and key-version ramp at the playground's scale and at the shop's (fluid model) ================
--- script coldstart.mjs
// Fluid model of a cold start under the design's limit: product page misses admitted
// at up to `budget` queries a second (300 - 60 checkout), each query costing `q` per page miss.
// Each admitted page miss fills its product. Cache capacity ignored until full (LRU cap).
function run({n, s=0.9, reads=1000, budget=240, q=1, cap, dt=0.5, tmax=6*3600}){
  // buckets
  let H=0; for(let i=1;i<=n;i++) H+=1/Math.pow(i,s);
  const B=[]; let st=1; while(st<=n){const e=Math.min(n, st<200?st:Math.max(st,Math.floor(st*1.01))); let w=0; for(let i=st;i<=e;i++) w+=1/Math.pow(i,s); B.push({c:e-st+1, r:reads*w/H/(e-st+1), x:0}); st=e+1;}
  let t=0, out={}, failSum=0, pagesSum=0; const marks=[10,30,60,300,600,1800,3600,7200,14400];
  while(t<tmax){
    let miss=0, keys=0; for(const b of B){miss+=b.c*b.r*(1-b.x); keys+=b.c*b.x;}
    const adm=Math.min(1, budget/(q*miss));
    failSum += miss*(1-adm)*dt; pagesSum+=reads*dt;
    if(keys<cap) for(const b of B){ b.x += b.r*adm*(1-b.x)*dt; if(b.x>1) b.x=1; }
    t+=dt;
    for(const m of marks) if(Math.abs(t-m)<dt/2) out[m]={failShare:+(miss*(1-adm)/reads).toFixed(3), hit:+(1-miss/reads).toFixed(3)};
    if(out.clear===undefined && miss*q<=budget) out.clear=t;
    if(out.budget===undefined && miss*q<=150) out.budget=t;
  }
  out.failedPages=Math.round(failSum); return out;
}

--- script ramp.mjs
// Fluid model of a key-version ramp (5% -> 100% of reads over `ramp` s) with the
// checkout-first limit (240 product queries a second), q queries per page miss.
function run({n, s=0.9, reads=1000, budget=240, q=1, ramp=120, dt=0.5, tmax=4*3600}){
  let H=0; for(let i=1;i<=n;i++) H+=1/Math.pow(i,s);
  const B=[]; let st=1; while(st<=n){const e=Math.min(n, st<200?st:Math.max(st,Math.floor(st*1.01))); let w=0; for(let i=st;i<=e;i++) w+=1/Math.pow(i,s); B.push({c:e-st+1, r:reads*w/H/(e-st+1), x:0}); st=e+1;}
  let t=0, fail=0, lastFail=0, peak=0;
  while(t<tmax){
    const share=Math.min(1, 0.05+0.95*t/ramp);
    let miss=0; for(const b of B) miss+=b.c*b.r*share*(1-b.x);
    miss += reads*(1-share)*0.05; // old version at ~95%
    peak=Math.max(peak, miss*q);
    const adm=Math.min(1, budget/(q*miss)); if(adm<1){fail+=miss*(1-adm)*dt; lastFail=t;}
    for(const b of B){ b.x += b.r*share*adm*(1-b.x)*dt; if(b.x>1)b.x=1; }
    t+=dt;
  }
  return {peakDemand:Math.round(peak), refusedUntil_s:lastFail, refusedPages:Math.round(fail)};
}
