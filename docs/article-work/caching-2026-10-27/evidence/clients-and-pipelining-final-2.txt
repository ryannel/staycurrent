# Final review 2: what redis-py 8.1 default retries re-send after a 100 ms timeout,
# with another client holding Redis in DEBUG SLEEP 0.5. usage: py_retry.py port [noretry]
import sys, threading, time
import redis
from redis.retry import Retry
from redis.backoff import NoBackoff
from redis.exceptions import WatchError

port = int(sys.argv[1])
noretry = len(sys.argv) > 2
kw = dict(port=port, socket_timeout=0.1)
if noretry:
    kw["retry"] = Retry(NoBackoff(), 0)
r = redis.Redis(**kw)
admin = redis.Redis(port=port)

def stall(delay=0.0):
    def go():
        time.sleep(delay)
        admin.execute_command("DEBUG", "SLEEP", "0.5")
    t = threading.Thread(target=go); t.start(); return t

def run(name, fn, per_run):
    admin.set("k", 0)
    r.ping()  # make sure the pooled connection is open before the stall
    t = stall(); time.sleep(0.02)
    t0 = time.monotonic()
    try:
        out = repr(fn())[:60]
    except Exception as e:
        out = type(e).__name__
    el = (time.monotonic() - t0) * 1000
    t.join(); time.sleep(0.3)
    v = int(admin.get("k"))
    print(f"{name:34s} ran {v // per_run} time(s)  reported {out}  after {el:.0f} ms")

def incr(): return r.incr("k")
def pipe_plain():
    p = r.pipeline(transaction=False); p.incr("k"); p.incr("k"); return p.execute()
def pipe_multi():
    p = r.pipeline(); p.incr("k"); p.incr("k"); return p.execute()
script = r.register_script("return redis.call('INCRBY', KEYS[1], 1)")
def lua(): return script(keys=["k"])

for name, fn, n in [("INCR", incr, 1), ("pipeline(transaction=False) 2xINCR", pipe_plain, 2),
                    ("pipeline() MULTI 2xINCR EXEC", pipe_multi, 2), ("Lua script +1", lua, 1)]:
    for _ in range(3):
        run(name, fn, n)

# WATCH loop: the stall begins after WATCH and GET, before MULTI ... EXEC.
def watch_incr_once():
    admin.set("k", 0)
    attempts = 0; errs = []
    t = None
    r.ping()
    p = r.pipeline()
    if True:
        while True:
            attempts += 1
            try:
                p.watch("k")
                v = int(p.get("k"))
                if attempts == 1:
                    t = stall(); time.sleep(0.02)
                p.multi(); p.set("k", v + 1); p.execute()
                break
            except WatchError as e:
                errs.append(str(e)[:50]); continue
            except Exception as e:
                errs.append(type(e).__name__); break
    if t: t.join()
    time.sleep(0.3)
    try: p.reset()
    except Exception: pass
    print(f"WATCH loop +1 (retry on WatchError)  k={admin.get('k').decode()}  attempts={attempts}  errors={errs}")

for _ in range(3):
    watch_incr_once()

# Final review 2: redis-py 8.1 WATCH loop, EXEC times out (100 ms) during a short stall.
# usage: py_watch.py port stall_seconds [noretry]
import sys, threading, time, redis
from redis.retry import Retry
from redis.backoff import NoBackoff
from redis.exceptions import WatchError
port=int(sys.argv[1]); st=sys.argv[2]
kw=dict(port=port, socket_timeout=0.1)
if len(sys.argv)>3: kw["retry"]=Retry(NoBackoff(),0)
r=redis.Redis(**kw); admin=redis.Redis(port=port)
for run in range(5):
    admin.set("k",0); r.ping(); errs=[]; attempts=0; t=None
    p=r.pipeline()
    while True:
        attempts+=1
        try:
            p.watch("k"); v=int(p.get("k"))
            if attempts==1:
                t=threading.Thread(target=lambda: admin.execute_command("DEBUG","SLEEP",st)); t.start(); time.sleep(0.02)
            p.multi(); p.set("k",v+1); p.execute(); break
        except WatchError as e: errs.append("WatchError: "+str(e)[:45]); continue
        except Exception as e: errs.append(type(e).__name__); break
    t.join(); time.sleep(0.3)
    try: p.reset()
    except Exception: pass
    print(f"stall {st}s: one +1 intended; k={admin.get('k').decode()} attempts={attempts} {errs}")

// Final review 2: what go-redis v9.23 default retries re-send after a 100 ms read timeout,
// with another client holding Redis in DEBUG SLEEP. usage: go run . port stall [noretry]
package main

import (
	"context"
	"errors"
	"fmt"
	"os"
	"strconv"
	"strings"
	"time"

	"github.com/redis/go-redis/v9"
)

func main() {
	ctx := context.Background()
	port := os.Args[1]
	stall := os.Args[2]
	opt := &redis.Options{Addr: "localhost:" + port, ReadTimeout: 100 * time.Millisecond}
	if len(os.Args) > 3 {
		opt.MaxRetries = -1
	}
	r := redis.NewClient(opt)
	admin := redis.NewClient(&redis.Options{Addr: "localhost:" + port, ReadTimeout: -1})
	script := redis.NewScript("return redis.call('INCRBY', KEYS[1], 1)")
	cases := []struct {
		name string
		per  int
		fn   func() (string, error)
	}{
		{"INCR", 1, func() (string, error) { v, err := r.Incr(ctx, "k").Result(); return fmt.Sprint(v), err }},
		{"Pipelined 2xINCR", 2, func() (string, error) {
			_, err := r.Pipelined(ctx, func(p redis.Pipeliner) error { p.Incr(ctx, "k"); p.Incr(ctx, "k"); return nil })
			return "", err
		}},
		{"TxPipelined MULTI 2xINCR EXEC", 2, func() (string, error) {
			_, err := r.TxPipelined(ctx, func(p redis.Pipeliner) error { p.Incr(ctx, "k"); p.Incr(ctx, "k"); return nil })
			return "", err
		}},
		{"Script.Run +1", 1, func() (string, error) { v, err := script.Run(ctx, r, []string{"k"}).Result(); return fmt.Sprint(v), err }},
		{"Watch loop +1 (retry on TxFailedErr)", 1, func() (string, error) {
			attempts := 0
			for {
				attempts++
				err := r.Watch(ctx, func(tx *redis.Tx) error {
					v, err := tx.Get(ctx, "k").Int()
					if err != nil {
						return err
					}
					if attempts == 1 {
						go admin.Do(ctx, "DEBUG", "SLEEP", stall)
						time.Sleep(20 * time.Millisecond)
					}
					_, err = tx.TxPipelined(ctx, func(p redis.Pipeliner) error { p.Set(ctx, "k", v+1, 0); return nil })
					return err
				}, "k")
				if errors.Is(err, redis.TxFailedErr) {
					continue
				}
				return fmt.Sprintf("attempts=%d", attempts), err
			}
		}},
	}
	for _, c := range cases {
		for i := 0; i < 3; i++ {
			admin.Set(ctx, "k", 0, 0)
			var held []*redis.Conn // warm the pool: hold 6 connections open at once, then return them
			for j := 0; j < 6; j++ {
				cn := r.Conn()
				cn.Ping(ctx)
				held = append(held, cn)
			}
			for _, cn := range held {
				cn.Close()
			}
			if !strings.HasPrefix(c.name, "Watch") {
				go admin.Do(ctx, "DEBUG", "SLEEP", stall)
				time.Sleep(20 * time.Millisecond)
			}
			t0 := time.Now()
			out, err := c.fn()
			el := time.Since(t0)
			time.Sleep(700 * time.Millisecond)
			v, _ := admin.Get(ctx, "k").Int()
			rep := "success " + out
			if err != nil {
				rep = "error: " + err.Error()
			}
			fmt.Printf("%-38s ran %s time(s)  reported %s  after %d ms\n", c.name, strconv.FormatFloat(float64(v)/float64(c.per), 'f', -1, 64), rep, el.Milliseconds())
		}
	}
}
