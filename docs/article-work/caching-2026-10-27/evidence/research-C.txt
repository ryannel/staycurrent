=== O. XFetch model, 20000 cycles per row, seed 7 ===
# MODEL, not a Redis measurement. One hot key; requests are a Poisson process of rate r per second.
# A recompute takes delta seconds; the value it writes lives T seconds. A cycle runs from the moment a fresh
# value is installed until the first recompute that starts in that cycle completes.
# XFetch rule (Vattani, Chierichetti, Lowenstein 2015): recompute if  now - delta*beta*ln(rand()) >= expiry
# Without XFetch, a request recomputes only when the value has expired (a miss). Nothing coordinates duplicate recomputes.
import random, math, statistics
def cycle(rng, r, delta, T, beta, xfetch):
    # Poisson arrivals are memoryless, so skip ahead: the XFetch trigger probability before T-30*beta*delta is below 1e-13 per request
    s = max(0.0, T - 30 * beta * delta) if xfetch else T; starts = []; hard = 0
    while True:
        s += rng.expovariate(r)
        if starts and s >= min(starts) + delta: break
        if s >= T:                       # value has expired: a hard miss, this request rebuilds
            hard += 1; starts.append(s)
        elif xfetch:
            u = rng.random() or 1e-300
            if s - delta * beta * math.log(u) >= T: starts.append(s)   # T == expiry offset
    return len(starts), hard, T - min(starts)
def run(r, delta, T, beta, n, seed, xfetch):
    rng = random.Random(seed); res = [cycle(rng, r, delta, T, beta, xfetch) for _ in range(n)]
    return (statistics.mean(x[0] for x in res), statistics.mean(x[1] for x in res),
            100 * sum(1 for x in res if x[1] > 0) / n, statistics.mean(x[2] for x in res), max(x[0] for x in res))

=== P. TTL jitter model, 10000 keys, seed 11 ===
# MODEL, not a Redis measurement. K keys are all loaded at t=0 (a deploy that warms the cache, or a flush).
# Each key is read as a Poisson process of rate lam per second. A read after expiry is a miss: it rebuilds the key
# (one database query) and installs it with TTL = base + Uniform(-J, +J). Count rebuilds per second for 3 hours.
import random, statistics, heapq
def sim(K, lam, base, J, horizon, seed):
    rng = random.Random(seed); per = [0]*(int(horizon)+2)
    # event: key expires at time e; next read after e arrives at e + Exp(lam) (memoryless), rebuild then, new expiry = that + ttl
    h = []
    for k in range(K): heapq.heappush(h, (base + (rng.uniform(-J, J) if J else 0), k))
    while h:
        e, k = heapq.heappop(h); t = e + rng.expovariate(lam)
        if t >= horizon: continue
        per[int(t)] += 1
        heapq.heappush(h, (t + base + (rng.uniform(-J, J) if J else 0), k))
    return per
K, lam, base, H = 10000, 1.0, 300, 3*3600
print("K=%d keys, each read about %.1f times per second, TTL base %d s, horizon %d s. Long-run mean rebuild rate = K/(base+1/lam) = %.1f per second" % (K, lam, base, H, K/(base+1/lam)))
print("jitter J (s) | peak rebuilds/s in first 15 min | peak in last hour | mean over last hour | max/mean in last hour | first-wave prediction K/(2J) rebuilds/s")
for J in [0, 15, 30, 60, 150]:
    per = sim(K, lam, base, J, H, 11)
    first = per[:900]; last = per[2*3600:3*3600]
    print("%5d | %6d | %6d | %7.1f | %5.1f | %s" % (J, max(first), max(last), statistics.mean(last), max(last)/statistics.mean(last), ("%.0f" % (K/(2*J))) if J else "-"))

import socket
PORT = 7203
class R:
    def __init__(self):
        self.s = socket.create_connection(("127.0.0.1", PORT)); self.s.setsockopt(socket.IPPROTO_TCP, socket.TCP_NODELAY, 1); self.f = self.s.makefile("rb")
    def _enc(self, *a):
        a = [x if isinstance(x, bytes) else str(x).encode() for x in a]
        return b"*%d\r\n" % len(a) + b"".join(b"$%d\r\n%s\r\n" % (len(x), x) for x in a)
    def _read(self):
        l = self.f.readline().rstrip(b"\r\n"); t = l[:1]; v = l[1:]
        if t == b"+": return v.decode()
        if t == b"-": raise RuntimeError(v.decode())
        if t == b":": return int(v)
        if t == b"$":
            n = int(v)
            if n < 0: return None
            return self.f.read(n + 2)[:-2].decode()
        if t == b"*":
            n = int(v); return None if n < 0 else [self._read() for _ in range(n)]
        raise RuntimeError("bad " + repr(l))
    def call(self, *a):
        self.s.sendall(self._enc(*a)); return self._read()
    def pipeline(self, cmds):
        self.s.sendall(b"".join(self._enc(*c) for c in cmds)); return [self._read() for _ in cmds]
# safety: refuse to run against any server that is not the group C server

=== Z. Cold-start model: how long an empty cache takes to refill when the database can only answer MU misses per second (illustrative; Zipf exponent and sizes are assumptions) ===
# MODEL, not a Redis measurement. After a flush the cache is empty. Requests arrive at LAMBDA per second for keys drawn
# from a Zipf distribution with exponent ALPHA over N keys. The database can answer at most MU queries per second; each answered
# miss fills the cache (no TTL expiry, infinite capacity, one fill per distinct key per second). Misses beyond MU get no answer and fill nothing.
# Output: seconds until the hit ratio first reaches 90% and 95% (measured over a 10 s window), and the database queries needed per second.
import numpy as np
def run(N, ALPHA, LAMBDA, MU, horizon, seed, steady_hit_note=False):
    rng = np.random.default_rng(seed)
    p = 1.0 / np.arange(1, N+1) ** ALPHA; p /= p.sum(); cdf = np.cumsum(p)
    cached = np.zeros(N, bool); hist = []; t90 = t95 = None
    for t in range(horizon):
        keys = np.searchsorted(cdf, rng.random(LAMBDA)); keys[keys >= N] = N-1
        hit = cached[keys]; miss_keys = keys[~hit]
        # database answers the first MU misses (arrival order); fills those keys
        answered = np.unique(miss_keys[:MU]); cached[answered] = True
        hist.append(hit.mean())
        if t >= 9:
            w = np.mean(hist[-10:])
            if t90 is None and w >= 0.90: t90 = t
            if t95 is None and w >= 0.95: t95 = t
            if t95 is not None: break
    return hist, t90, t95, (cached.sum())
print("N keys, Zipf alpha, requests/s, db queries/s | hit ratio in first second | seconds to 90% | seconds to 95% | keys cached at the end")
for (N, a, lam, mu) in [(1_000_000, 1.0, 3000, 300), (1_000_000, 0.8, 3000, 300), (100_000, 1.0, 3000, 300), (1_000_000, 1.0, 3000, 3000), (1_000_000, 1.0, 3000, 100)]:
    h, t90, t95, nc = run(N, a, lam, mu, 7200, 5)
    print("%9d %4.1f %5d %5d | %.3f | %s | %s | %d" % (N, a, lam, mu, h[0], t90, t95, nc))
