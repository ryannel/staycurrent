# Minimal RESP2 client for the review (port 7157 only). No third-party packages.
import socket
class R:
    def __init__(self, port=7157):
        self.s = socket.create_connection(("127.0.0.1", port)); self.s.setsockopt(socket.IPPROTO_TCP, socket.TCP_NODELAY, 1)
        self.f = self.s.makefile("rb")
    def send(self, *args):
        out = [b"*%d\r\n" % len(args)]
        for a in args:
            a = a if isinstance(a, bytes) else str(a).encode()
            out.append(b"$%d\r\n%s\r\n" % (len(a), a))
        self.s.sendall(b"".join(out))
    def read(self):
        line = self.f.readline()
        t, rest = line[:1], line[1:-2]
        if t == b"+": return rest.decode()
        if t == b"-": return "ERR:" + rest.decode()
        if t == b":": return int(rest)
        if t == b"$":
            n = int(rest)
            if n < 0: return None
            d = self.f.read(n + 2)[:-2]
            try: return d.decode()
            except UnicodeDecodeError: return d
        if t == b"*":
            n = int(rest)
            return None if n < 0 else [self.read() for _ in range(n)]
        raise ValueError(line)
    def __call__(self, *args):
        self.send(*args); return self.read()
    def info(self, section):
        d = {}
        for l in self("INFO", section).splitlines():
            if ":" in l and not l.startswith("#"):
                k, v = l.split(":", 1); d[k] = v
        return d
