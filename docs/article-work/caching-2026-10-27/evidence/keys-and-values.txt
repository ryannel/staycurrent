## big.py
# Big values: transfer time of a large string, and how a large delete holds up another client.
# Raw RESP over loopback to port 7103. Python 3, no third-party modules.
import socket, time, threading
def conn():
    s = socket.create_connection(("127.0.0.1", 7103)); s.setsockopt(socket.IPPROTO_TCP, socket.TCP_NODELAY, 1); return s
def enc(*args):
    out = [b"*%d\r\n" % len(args)]
    for a in args:
        a = a if isinstance(a, bytes) else str(a).encode()
        out.append(b"$%d\r\n%s\r\n" % (len(a), a))
    return b"".join(out)
def read_reply(s, buf=bytearray()):
    # minimal reader: returns once a full reply for simple/int/bulk types is read
    f = s.makefile("rb")
    line = f.readline()
    t = line[:1]
    if t in b"+-:": return line
    if t == b"$":
        n = int(line[1:-2])
        if n < 0: return None
        data = f.read(n + 2); return data[:-2]
    if t == b"*":
        n = int(line[1:-2]); return [read_one(f) for _ in range(n)]
    return line
def read_one(f):
    line = f.readline(); t = line[:1]
    if t == b"$":
        n = int(line[1:-2]); return None if n < 0 else f.read(n + 2)[:-2]
    return line
def cmd(s, *args):
    s.sendall(enc(*args)); return read_reply(s)

a = conn(); b = conn()
cmd(a, "FLUSHALL"); cmd(a, "CONFIG", "SET", "slowlog-log-slower-than", "1000"); cmd(a, "SLOWLOG", "RESET")
for size in (1024, 100*1024, 1024*1024, 20*1024*1024):
    v = b"v" * size
    t = time.perf_counter(); cmd(a, "SET", "blob", v); ts = time.perf_counter() - t
    times = []
    for _ in range(5):
        t = time.perf_counter(); r = cmd(a, "GET", "blob"); times.append(time.perf_counter() - t)
        assert len(r) == size
    times.sort()
    print(f"  value {size:>9,} bytes: SET {ts*1000:7.2f} ms, GET median of 5 {times[2]*1000:7.2f} ms (loopback)")

# Build a hash with 1,000,000 fields in batches.
def build(name):
    for start in range(0, 1_000_000, 10_000):
        args = ["HSET", name]
        for i in range(start, start + 10_000): args += [f"f{i}", "v"]
        cmd(a, *args)
build("bighash")
print("  MEMORY USAGE bighash SAMPLES 0:", cmd(a, "MEMORY", "USAGE", "bighash", "SAMPLES", "0").decode().strip())
print("  HLEN bighash:", cmd(a, "HLEN", "bighash").decode().strip())
# Second client sends PING every 2 ms; record the worst wait while the first client deletes the hash.
lat = []; stop = False
def pinger():
    while not stop:
        t = time.perf_counter(); cmd(b, "PING"); lat.append((t, time.perf_counter() - t)); time.sleep(0.002)

# Loop GET of key argv[1] for argv[2] seconds, reading each full reply.
import socket, sys, time
key = sys.argv[1].encode(); secs = float(sys.argv[2]); s = socket.create_connection(('127.0.0.1', 7113))

    s.sendall(cmd); head = b''
    while not head.endswith(b'\r\n'): head += s.recv(1)
