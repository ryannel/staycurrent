# ---- rtools.py ----
# Small Redis helpers for article 7 evidence. Plain sockets, RESP2, port 7107 only.
import socket, time, threading, statistics, sys, subprocess
PORT = 7107
def resp(*a):
    b = b"*%d\r\n" % len(a)
    for x in a:
        x = x if isinstance(x, bytes) else str(x).encode()
        b += b"$%d\r\n%s\r\n" % (len(x), x)
    return b
class C:
    def __init__(s):
        s.s = socket.create_connection(("127.0.0.1", PORT)); s.s.setsockopt(socket.IPPROTO_TCP, socket.TCP_NODELAY, 1); s.f = s.s.makefile("rb")
    def read(s):
        line = s.f.readline(); t = line[:1]
        if t == b"+": return line[1:-2].decode()
        if t == b"-": return "ERR:" + line[1:-2].decode()
        if t == b":": return int(line[1:-2])
        if t == b"$":
            n = int(line[1:-2]); return None if n < 0 else s.f.read(n + 2)[:-2]
        if t == b"*":
            n = int(line[1:-2]); return None if n < 0 else [s.read() for _ in range(n)]
        raise Exception(line)
    def do(s, *a):
        s.s.sendall(resp(*a)); return s.read()
def pipe_load(gen):
    """Mass insert with redis-cli --pipe; gen yields RESP bytes."""
    p = subprocess.Popen(["/opt/homebrew/opt/redis/bin/redis-cli", "-p", str(PORT), "--pipe"], stdin=subprocess.PIPE, stdout=subprocess.PIPE, stderr=subprocess.STDOUT)
    buf = []
    for chunk in gen:
        buf.append(chunk)
        if len(buf) >= 10000: p.stdin.write(b"".join(buf)); buf = []
    if buf: p.stdin.write(b"".join(buf))
    p.stdin.close(); out = p.stdout.read().decode().strip().splitlines()[-1]; p.wait(); return out
class Prober:
    """A second client that sends PING, waits for the reply, sleeps ~1 ms, repeats; records round trips in ms with timestamps."""
    def __init__(s):
        s.lat = []; s.stop = False; s.c = C(); s.t0 = time.perf_counter()
        s.th = threading.Thread(target=s.run); s.th.start(); time.sleep(0.3)
    def run(s):
        while not s.stop:
            t = time.perf_counter(); s.c.s.sendall(b"PING\r\n"); s.c.s.recv(100)
            s.lat.append(((t - s.t0) * 1000, (time.perf_counter() - t) * 1000)); time.sleep(0.001)
    def mark(s): return len(s.lat)
    def finish(s): s.stop = True; s.th.join()
def summary(vals):
    v = sorted(vals)
    return "n=%d median=%.3f ms p99=%.3f ms max=%.3f ms" % (len(v), statistics.median(v), v[min(len(v)-1, int(len(v)*0.99))], v[-1])

# scanprobe.py COUNT [MATCH]: iterate the whole keyspace with SCAN ... COUNT while a second client PINGs every ~1 ms.

# tmpl.py: used_memory growth for 200,000 hashes of 5 named fields (20-byte values), with hash-min-template-entries 0 and 3.
import subprocess
R = ['redis-cli', '-p', '7117']
def r(*a): return subprocess.run(R + list(a), capture_output=True, text=True).stdout.strip()
def used(): return int([l for l in r('info', 'memory').splitlines() if l.startswith('used_memory:')][0].split(':')[1])
def resp(*args): return ''.join([f'*{len(args)}\r\n'] + [f'${len(a.encode())}\r\n{a}\r\n' for a in args])
names = ['title_en', 'price_cents', 'currency', 'stock', 'brand']
for setting in ['0', '3']:
    r('flushall'); r('config', 'set', 'hash-min-template-entries', setting)
    before = used()
    data = ''.join(resp('HSET', f'product:{i:08d}', *[x for n in names for x in (n, f'{i:020d}')]) for i in range(200000))
    subprocess.run(R + ['--pipe'], input=data, capture_output=True, text=True)
    after = used()
    print(f'hash-min-template-entries {setting}: encoding {r("object", "encoding", "product:00000005")}, '
          f'{(after - before) / 200000:.1f} bytes per hash (used_memory growth / 200,000), MEMORY USAGE {r("memory", "usage", "product:00000005")}')
r('config', 'set', 'hash-min-template-entries', '0'); r('flushall')
