## B. Race 1: a slow reader refills an old value after the writer's delete
# Database holds price 39. Reader misses and reads 39 from the database (not shown: the database is the application's).
> GET product:42
  
# Writer commits 35 to the database, then deletes the cached copy. There is nothing to delete yet.
> DEL product:42
  0
# The reader resumes and stores the 39 it read earlier, with a one-hour TTL.
> SET product:42 {"price":39} EX 3600
  OK
> GET product:42
  {"price":39}
> TTL product:42
  3600
# A second delete (for example from the next write) is the only thing besides expiry that removes it.
> DEL product:42
  1

## C. Race 2: two writers update the cache out of order
> SET product:42 {"price":39} EX 3600
  OK
# Database order: W1 commits 37, then W2 commits 35. Their cache writes arrive W2 first, then W1.
> SET product:42 {"price":35} EX 3600
  OK
> SET product:42 {"price":37} EX 3600
  OK
> GET product:42
  {"price":37}
# With delete-on-write the same reordering is harmless: two deletes in either order leave the key absent.
> DEL product:42
  1
> DEL product:42
  0
> EXISTS product:42
  0

## D. SET ... IFEQ semantics (Redis 8.4+)
> SET k old EX 100
  OK
> SET k new IFEQ wrong
  
> GET k
  old
> SET k new IFEQ old
  OK
> GET k
  new
> TTL k
  -1
# IFEQ is an ordinary SET: without EX or KEEPTTL the old TTL is dropped (-1 above). With EX:
> SET k newer IFEQ new EX 100
  OK
> TTL k
  100
> SET k newest IFEQ newer KEEPTTL
  OK
> TTL k
  100
# On a missing key IFEQ fails:
> DEL k
  1
> SET k value IFEQ anything
  
> EXISTS k
  0
# IFEQ with GET returns the old value:
> SET k a
  OK
> SET k b IFEQ a GET
  a
> GET k
  b
# DELEX compare-and-delete:
> DELEX k IFEQ a
  0
> DELEX k IFEQ b
  1
> EXISTS k
  0
# NX with a lease placeholder:
> SET k lease:7f3a NX PX 10000
  OK
> SET k lease:9c21 NX PX 10000
  
> GET k
  lease:7f3a
> DEL k
  1

## E. A lease held in the key itself (placeholder plus SET ... IFEQ, Redis 8.4+)
# Reader R misses, then takes the lease by writing a placeholder with a unique token, before reading the database.
> GET product:42
  
> SET product:42 lease:r-81c2 NX PX 10000
  OK
# A second reader S misses too. Its NX fails: someone is already loading this key.
> GET product:42
  lease:r-81c2
> SET product:42 lease:s-4d07 NX PX 10000
  
# R reads 39 from the database and pauses. Writer W commits 35 and deletes the key, which removes R's placeholder.
> DEL product:42
  1
# R resumes and fills only if its own placeholder is still there. It is not, so the old value is refused.
> SET product:42 {"price":39} IFEQ lease:r-81c2 EX 3600
  
> EXISTS product:42
  0
# The next reader takes a new lease, reads 35 and fills.
> SET product:42 lease:t-0b9e NX PX 10000
  OK
> SET product:42 {"price":35} IFEQ lease:t-0b9e EX 3600
  OK
> GET product:42
  {"price":35}
> TTL product:42
  3600
# Without a racing write, the fill succeeds:
> DEL product:42
  1
> SET product:42 lease:u-11aa NX PX 10000
  OK
> SET product:42 {"price":35} IFEQ lease:u-11aa EX 3600
  OK
> GET product:42
  {"price":35}
# A lease that expires (PX 200, filler takes 400 ms) also refuses the late fill:
> DEL product:42
  1
> SET product:42 lease:v-5e5e NX PX 200
  OK
> SET product:42 {"price":35} IFEQ lease:v-5e5e EX 3600
  
> EXISTS product:42
  0
# A late fill cannot overwrite a newer reader's lease either (tokens differ):
> SET product:42 lease:w-2222 NX PX 10000
  OK
> SET product:42 {"price":39} IFEQ lease:v-5e5e EX 3600
  
> GET product:42
  lease:w-2222
> DEL product:42
  1

## G. Set only if newer built from SET ... IFEQ (Redis 8.4+): read, compare in the application, write if unchanged
> SET product:42 v7|{"price":37} EX 3600
  OK
# Writer A reads the current value, sees version 7, and wants to write version 9.
> GET product:42
  v7|{"price":37}
# Meanwhile writer B writes version 8 with the same method:
> SET product:42 v8|{"price":36} IFEQ v7|{"price":37} EX 3600
  OK
# A's write is conditional on the value it read; it fails, so A reads again, compares 9 > 8, and retries:
> SET product:42 v9|{"price":35} IFEQ v7|{"price":37} EX 3600
  
> GET product:42
  v8|{"price":36}
> SET product:42 v9|{"price":35} IFEQ v8|{"price":36} EX 3600
  OK
> GET product:42
  v9|{"price":35}
# A late writer with version 8 reads v9, sees 8 < 9, and writes nothing.
> DEL product:42
  1

## F2. Versioned values with tombstones: the script used in the article
# A value is a hash {version, value}; a tombstone is a hash {version} with no value field.
# Write if the cache has nothing, or an older version, or a tombstone of the same version.
local cur = tonumber(redis.call("HGET", KEYS[1], "version") or "-1")
local new = tonumber(ARGV[1])
local dead = redis.call("HEXISTS", KEYS[1], "value") == 0
if new > cur or (new == cur and dead and ARGV[2] ~= "") then
  redis.call("DEL", KEYS[1])
  if ARGV[2] == "" then redis.call("HSET", KEYS[1], "version", ARGV[1])
  else redis.call("HSET", KEYS[1], "version", ARGV[1], "value", ARGV[2]) end
  redis.call("EXPIRE", KEYS[1], ARGV[3])
  return 1
end
return 0

## Q. Race 1 on a real Redis with a simulated database (py/race_real.py)

# sleeps an exponential fill time, then SET race:42 <version> EX ttl. A writer commits (increments
# the version) then DEL race:42. A sampler GETs every 5 ms and compares with the database version.
# 60 s per run. The point is only that the race occurs with real commands; the counts are too
# small for rates, and the article's rates come from the model in race-model.mjs.
