const md5 = t => createHash('md5').update(t).digest().readUInt32BE(0);
function ringWith(hash, servers, points = 160) {
  const marks = [];
  for (let s = 0; s < servers; s++) for (let v = 0; v < points; v++) marks.push([hash(`server-${s}#${v}`), s]);
  marks.sort((a, b) => a[0] - b[0]);
  return key => { const h = hash(key); let i = marks.findIndex(m => m[0] >= h); if (i < 0) i = 0; return marks[i][1]; };
}
function moved(hash, n, count = 20000) {
  const a = ringWith(hash, n), b = ringWith(hash, n + 1); let m = 0;
  for (let i = 0; i < count; i++) { const k = `product:${i}`; if (a(k) !== b(k)) m++; }
  return m / count;
}
