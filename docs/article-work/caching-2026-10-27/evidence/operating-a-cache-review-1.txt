    import { connect } from './resp.mjs';
    const PORT = Number(process.argv[2]);
    const admin = await connect(PORT), probe = await connect(PORT);
    async function load() { await admin.call('FLUSHALL', 'SYNC'); const v = 'x'.repeat(100);
      for (let i = 0; i < 500000; i += 10000) { const b = []; for (let j = i; j < i + 10000; j++) b.push(admin.call('SET', 'k:' + j, v)); await Promise.all(b); }
      return admin.call('DBSIZE'); }
    async function run(label, fn) {
      const n = await load(); await new Promise(r => setTimeout(r, 200));
      let worst = 0, stop = false, count = 0;
      const p = (async () => { while (!stop) { const t = performance.now(); await probe.call('PING'); worst = Math.max(worst, performance.now() - t); count++; await new Promise(r => setTimeout(r, 1)); } })();
      await new Promise(r => setTimeout(r, 100)); worst = 0;
      const t0 = performance.now(); const reply = await fn(); const took = performance.now() - t0;
      await new Promise(r => setTimeout(r, 400)); stop = true; await p;
      console.log(`${label} on ${n} keys: reply ${JSON.stringify(reply)} after ${took.toFixed(1)} ms; worst PING from other client ${worst.toFixed(1)} ms (${count} pings)`);
    }
    for (let k = 0; k < 2; k++) {
      await run('FLUSHALL (plain)', () => admin.call('FLUSHALL'));
      await run('FLUSHALL SYNC', () => admin.call('FLUSHALL', 'SYNC'));
      await run('FLUSHALL ASYNC', () => admin.call('FLUSHALL', 'ASYNC'));
      await run('MULTI; FLUSHALL; EXEC', async () => { admin.call('MULTI'); admin.call('FLUSHALL'); return admin.call('EXEC'); });
