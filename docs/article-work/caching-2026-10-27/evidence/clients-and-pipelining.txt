Evidence for "Redis clients: round trips, pipelining and atomic operations" (article 8)
Gathered 2026-10-27.

Environment
  Redis 8.10.3 (Homebrew build, macOS, libc malloc, kqueue), Apple M3 Max laptop.
  Server started for these experiments only:
    /opt/homebrew/opt/redis/bin/redis-server --port 7108 --dir <temporary directory> --save '' --appendonly no
  It was restarted once on purpose (E12 ends with SHUTDOWN NOSAVE; E13 restarts it to show that
  functions and the script cache do not survive a restart without persistence), and shut down
  with SHUTDOWN NOSAVE when the experiments were finished.
  Clients: Python 3.14 with a minimal RESP2 client (r8.py, standard library only), on the same
  machine over loopback with TCP_NODELAY; E18 uses redis-py 8.1.0 from PyPI with its defaults.
  E2-proxy runs through delayproxy.py, which holds every chunk of bytes for 0.5 ms in each
  direction (a measured round trip of about 1.56 ms).
  The machine was shared with other agents during the runs: load averages between about 18 and
  34 on 16 cores. Absolute timings are therefore noisy; the comparisons were run back to back.

==== Appendix: script sources

---- r8.py
# Minimal RESP2 client for the article 8 experiments (standard library only).
import socket, time, statistics

PORT = 7108
HOST = "127.0.0.1"


def enc(*args):
    out = [b"*%d\r\n" % len(args)]
    for a in args:
        a = a if isinstance(a, bytes) else str(a).encode()
        out.append(b"$%d\r\n%s\r\n" % (len(a), a))
    return b"".join(out)


class Conn:
    def __init__(self, port=PORT, host=HOST, timeout=None):
        self.s = socket.create_connection((host, port))
        self.s.setsockopt(socket.IPPROTO_TCP, socket.TCP_NODELAY, 1)
        if timeout is not None:
            self.s.settimeout(timeout)
        self.buf = b""
        self.pos = 0

    def close(self):
        self.s.close()

    def send(self, data):
        self.s.sendall(data)

    def _fill(self):
        d = self.s.recv(1 << 20)
        if not d:
            raise ConnectionError("closed")
        if self.pos:
            self.buf = self.buf[self.pos:]
            self.pos = 0
        self.buf += d

    def _line(self):
        i = self.buf.find(b"\r\n", self.pos)
        while i < 0:
            self._fill()
            i = self.buf.find(b"\r\n", self.pos)
        line = self.buf[self.pos:i]
        self.pos = i + 2
        return line

    def read(self):
        line = self._line()
        t, rest = line[:1], line[1:]
        if t == b"+":
            return rest.decode()
        if t == b"-":
            return Err(rest.decode())
        if t == b":":
            return int(rest)
        if t == b"$":
            n = int(rest)
            if n < 0:
                return None
            while len(self.buf) - self.pos < n + 2:
                self._fill()
            v = self.buf[self.pos:self.pos + n]
            self.pos += n + 2
            return v
        if t == b"*":
            n = int(rest)
            if n < 0:
                return None
            return [self.read() for _ in range(n)]
        if t == b"_":
            return None
        if t == b"%":
            n = int(rest)
            return {self.read(): self.read() for _ in range(n)}
        return ("?", line)

    def cmd(self, *args):
        self.send(enc(*args))
        return self.read()

    def pipeline(self, cmds):
        self.send(b"".join(enc(*c) for c in cmds))
        return [self.read() for _ in cmds]


class Err(str):

---- delayproxy.py
# A TCP proxy that holds every chunk of bytes for a fixed delay in each direction,
# to imitate a network round trip on one machine. usage: delayproxy.py listen_port target_port one_way_ms
import asyncio, sys
LISTEN, TARGET, DELAY = int(sys.argv[1]), int(sys.argv[2]), float(sys.argv[3]) / 1000
async def pump(reader, writer):
    loop = asyncio.get_running_loop()
    try:
        while True:
            data = await reader.read(1 << 20)
            if not data: break
            loop.call_later(DELAY, writer.write, data)   # constant delay keeps order
    finally:
        await asyncio.sleep(DELAY + 0.01)
        writer.close()
async def handle(cr, cw):
    sr, sw = await asyncio.open_connection("127.0.0.1", TARGET)
    for w in (cw, sw):
        sock = w.get_extra_info("socket"); import socket; sock.setsockopt(socket.IPPROTO_TCP, socket.TCP_NODELAY, 1)
    await asyncio.gather(pump(cr, sw), pump(sr, cw))
async def main():
    srv = await asyncio.start_server(handle, "127.0.0.1", LISTEN)
    async with srv: await srv.serve_forever()
asyncio.run(main())

	"time"

const addr = "127.0.0.1:7118"

var ctx = context.Background()
var admin = redis.NewClient(&redis.Options{Addr: addr})

func warm(c *redis.Client) {
	var wg sync.WaitGroup
	for j := 0; j < 6; j++ {
		wg.Add(1)
		go func() { defer wg.Done(); c.Do(ctx, "DEBUG", "SLEEP", "0.01") }()
	}
	wg.Wait()
}
func stall() {
	go func() {
		s := redis.NewClient(&redis.Options{Addr: addr, ReadTimeout: 10 * time.Second})
		s.Do(ctx, "DEBUG", "SLEEP", "0.5")
		s.Close()
	}()
	time.Sleep(20 * time.Millisecond)
}

var script = redis.NewScript(`return redis.call('INCRBY', KEYS[1], 1)`)

func run(name string, maxRetries int, f func(c *redis.Client) (string, error)) {
	for i := 0; i < 2; i++ {
		admin.Set(ctx, "a", 0, 0)
		admin.Set(ctx, "b", 0, 0)
		c := redis.NewClient(&redis.Options{Addr: addr, ReadTimeout: 100 * time.Millisecond, MaxRetries: maxRetries})
		warm(c)
		script.Load(ctx, c)
		stall()
		t := time.Now()
		res, err := f(c)
		el := time.Since(t).Round(time.Millisecond)
		time.Sleep(800 * time.Millisecond)
		fmt.Printf("%-40s MaxRetries=%2d: result=%s err=%v after %v; a=%s b=%s\n", name, maxRetries, res, err, el, admin.Get(ctx, "a").Val(), admin.Get(ctx, "b").Val())
		c.Close()
	}
}

		run("INCR a", mr, func(c *redis.Client) (string, error) {

			return fmt.Sprint(v), err
		})

			return fmt.Sprint(len(cmds)), err
		})
		run("EVALSHA script INCRBY a 1", mr, func(c *redis.Client) (string, error) {

		})

import sys, threading, time, uuid, redis
from redis.retry import Retry
from redis.backoff import NoBackoff
from redis.exceptions import WatchError

port = int(sys.argv[1]); stall = sys.argv[2]
admin = redis.Redis(port=port)

def client(noretry):
    kw = dict(port=port, socket_timeout=0.1)
    if noretry:
        kw["retry"] = Retry(NoBackoff(), 0)
    return redis.Redis(**kw)

def start_stall():
    t = threading.Thread(target=lambda: admin.execute_command("DEBUG", "SLEEP", stall))
    t.start(); time.sleep(0.02); return t

def run(label, r, body, check):
    for _ in range(5):
        admin.delete("k", "applied:req1"); admin.hset("k", mapping={"version": 8, "n": 0})
        r.ping()
        st = {"attempts": 0, "t": None}

            st["attempts"] += 1
            stop = body(p)            # reads, then decides; returns True to stop without writing
            if st["attempts"] == 1:
                st["t"] = start_stall()
            if stop:
                p.unwatch(); return
            p.multi(); write(p)
        global write
        out = "ok"
        try:
            r.transaction(func, "k")
        except Exception as e:
            out = type(e).__name__
        if st["t"]: st["t"].join()
        time.sleep(0.3)
        print(f"{label:58s} stall {stall}s: attempts={st['attempts']} raised={out:12s} {check()}")
