=== 2. Flush race with ONE persistent connection (python socket flusher) and redis-benchmark -c 20 -n 200000 INCR
# Flusher over one persistent connection (not a process per command).
# mode 'getdel': GETDEL views:x ; mode 'split': GET views:x then DEL views:x
import socket, sys, time
mode = sys.argv[1]; stop_file = sys.argv[2]
s = socket.create_connection(("127.0.0.1", 7152)); f = s.makefile("rb")
def cmd(*a):
    s.sendall(b"*%d\r\n" % len(a) + b"".join(b"$%d\r\n%s\r\n" % (len(x), x) for x in (y.encode() for y in a)))
    line = f.readline()
    if line[:1] == b"$":
        n = int(line[1:]); 
        if n < 0: return None
        v = f.read(n + 2)[:-2]; return v
    return line
total = 0; import os
while True:
    if mode == "getdel":
        v = cmd("GETDEL", "views:x")
    else:
        v = cmd("GET", "views:x")
        if v is not None: cmd("DEL", "views:x")
    if v is not None: total += int(v)
    elif os.path.exists(stop_file): break
print(total)
#!/bin/bash
S=$1; mode=$2
R=(/opt/homebrew/opt/redis/bin/redis-cli -p 7152)

touch $S/stop; wait $FP
rest=$("${R[@]}" GET views:x); 
echo "mode=$mode increments=200000 flushed=$(cat $S/flushed.$mode) left_in_redis=${rest:-0}"

=== 6. Delete-on-write hit ratio under different arrival patterns (pattern.mjs)
// Delete-on-write hit ratio for one key with k reads per write, under three arrival patterns.
function run(seq){ let cached=true,r=0,h=0; for(const e of seq){ if(e==='W') cached=false; else { r++; if(cached) h++; else cached=true; } } return h/r; }
let seed=7; const rnd=()=>((seed=(seed*1103515245+12345)%2147483648)/2147483648);
for (const k of [1,2,4,19]) {
  const N=200000;
  const even=[]; for(let i=0;i<N;i++){ for(let j=0;j<k;j++) even.push('R'); even.push('W'); }
  const poisson=[]; for(let i=0;i<N*(k+1);i++) poisson.push(rnd()<1/(k+1)?'W':'R');
  const bursty=[]; for(let i=0;i<N/5;i++){ for(let j=0;j<5*k;j++) bursty.push('R'); for(let j=0;j<5;j++) bursty.push('W'); }
  const rmw=[]; for(let i=0;i<N;i++){ for(let j=0;j<k-1;j++) rmw.push('R'); rmw.push('W'); rmw.push('R'); }
  console.log(`k=${k}: formula k/(k+1)=${(k/(k+1)*100).toFixed(1)}%  random=${(run(poisson)*100).toFixed(1)}%  evenly spaced=${(run(even)*100).toFixed(1)}%  writes in bursts of 5=${(run(bursty)*100).toFixed(1)}%  read-own-write after each write=${(run(rmw)*100).toFixed(1)}%`);
}

=== 8. Primary sources checked (quotes)
Bronson et al. HotOS 2021 §2.2: 'the cache will remain empty since the web application is responsible for populating the cache, but its timeout will cause all queries to be considered as failed.'
Bronson §3: 'This prioritization is unenforceable with a look-aside cache but trivial with a read-through cache. A read-through cache can have a permissive timeout for database queries'
Nishtala et al. NSDI 2013 §2: 'demand-filled look-aside cache'; 'We choose to delete cached data instead of updating it because deletes are idempotent.' §3.1: 'Clients treat get errors as cache misses, but web servers will skip inserting entries into memcached after querying for data to avoid putting additional load on a possibly overloaded network or server.' (context: UDP get drops)
