== E2 MULTI/EXEC and WATCH semantics (tx.py, separate connections, retries off) ==
## MULTI with queue-time error
('MULTI',) -> b'OK'
('INCR', 'x') -> b'QUEUED'
('INCRR', 'x') -> ERR<ResponseError: unknown command 'INCRR', with args beginning with: 'x' >
('EXEC',) -> ERR<ExecAbortError: Transaction discarded because of previous errors.>
x = b'0'
## MULTI with run-time error
('EXEC',) -> [1, server:ResponseError, 2]
## WATCH: other client writes same value              EXEC -> None k= b'v'
## WATCH: other client writes a different key         EXEC -> [b'OK'] k= b'mine'
## WATCH: watched key expires before EXEC             EXEC -> None
## WATCH: other client's GET (read) of the key        EXEC -> [b'OK']
## WATCH: other client's EXPIRE on the key            EXEC -> None
## WATCH: key evicted under maxmemory (allkeys-random) k exists after filling: 0 evicted_keys= 2863 ; EXEC -> None
## raw socket, one write: "WATCH k\r\nGET k\r\nMULTI\r\nSET k 9\r\nEXEC\r\n"
b'+OK\r\n$1\r\n8\r\n+OK\r\n+QUEUED\r\n*1\r\n+OK\r\n'   -> WATCH and GET can share a round trip; an uncontested WATCH write needs 2 round trips, not 3.
Docs (develop/using-commands/transactions): "This includes modifications made by the client, like write commands, and by Redis itself, like expiration or eviction."; "In Redis versions before 6.0.9, an expired key would not cause a transaction to be aborted."

import redis, time, threading, sys, statistics
from redis.retry import Retry
from redis.backoff import NoBackoff
P=7158
admin = redis.Redis(port=P)
def sleeper(sec):
    redis.Redis(port=P, socket_timeout=10).execute_command('DEBUG','SLEEP',str(sec))
def trial(cmd, retries_off=False):
    admin.set('views',100); admin.set('price',30)
    kw = dict(port=P, socket_timeout=0.1, socket_connect_timeout=0.1)
    if retries_off: kw['retry']=Retry(NoBackoff(),0)
    c = redis.Redis(**kw); c.ping()
    t = threading.Thread(target=sleeper, args=(0.5,)); t.start(); time.sleep(0.02)
    t0=time.perf_counter()
    try: r = c.incr('views') if cmd=='incr' else c.set('price',35); err=None
    except Exception as e: r=None; err=type(e).__name__
    el=(time.perf_counter()-t0)*1000
    t.join(); time.sleep(0.3)
    print(f"{cmd} retries_off={retries_off}: result={r} err={err} elapsed={el:.0f}ms views={admin.get('views')} price={admin.get('price')}")
for i in range(3): trial('incr')
for i in range(2): trial('set')
for i in range(2): trial('incr', True)

import redis, time, statistics
from redis.retry import Retry
from redis.backoff import NoBackoff
def t(host, port, n, retries_off=False, **kw):
    out=[]
    for i in range(n):
        args=dict(host=host, port=port, **kw)
        if retries_off: args['retry']=Retry(NoBackoff(),0)
        c=redis.Redis(**args); t0=time.perf_counter()

from redis.backoff import NoBackoff

    a=redis.Redis(port=P, socket_timeout=0.1); a.ping()            # default retry policy

    th.join(); tb.join(); time.sleep(0.2)

--- gotest/main.go
package main

import (
	"context"
	"fmt"
	"sync"
	"time"

	"github.com/redis/go-redis/v9"
)

func main() {
	ctx := context.Background()
	admin := redis.NewClient(&redis.Options{Addr: "127.0.0.1:7158"})
	for i := 0; i < 3; i++ {
		admin.Set(ctx, "views", 100, 0)
		c := redis.NewClient(&redis.Options{Addr: "127.0.0.1:7158", ReadTimeout: 100 * time.Millisecond})
		var wg sync.WaitGroup // warm the pool with 6 open connections
		for j := 0; j < 6; j++ { wg.Add(1); go func() { defer wg.Done(); c.Do(ctx, "DEBUG", "SLEEP", "0.01") }() }
		wg.Wait()
		fmt.Printf("pool before: %+v\n", c.PoolStats().TotalConns)
		go func() { s := redis.NewClient(&redis.Options{Addr: "127.0.0.1:7158", ReadTimeout: 10 * time.Second}); s.Do(ctx, "DEBUG", "SLEEP", "0.5"); s.Close() }()
		time.Sleep(20 * time.Millisecond)
		t := time.Now()
		v, err := c.Incr(ctx, "views").Result()
		el := time.Since(t).Round(time.Millisecond)
		time.Sleep(800 * time.Millisecond)
		fmt.Printf("go-redis v9.23 INCR, warm pool, 100ms ReadTimeout, 0.5s stall: result=%d err=%v after %v; views now %s\n", v, err, el, admin.Get(ctx, "views").Val())
		c.Close()
	}
}
