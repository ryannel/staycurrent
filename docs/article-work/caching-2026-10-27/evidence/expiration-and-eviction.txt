#### script resp.py
import socket
class R:
    def __init__(s,port=7104):
        s.s=socket.create_connection(("127.0.0.1",port)); s.s.setsockopt(socket.IPPROTO_TCP, socket.TCP_NODELAY,1); s.f=s.s.makefile('rb')
    @staticmethod
    def enc(args):
        o=[f"*{len(args)}\r\n".encode()]
        for a in args:
            if isinstance(a,str): a=a.encode()
            elif isinstance(a,int): a=str(a).encode()
            o.append(b"$%d\r\n"%len(a)); o.append(a); o.append(b"\r\n")
        return b"".join(o)
    def _read(s):
        l=s.f.readline()
        t=l[:1]; b=l[1:-2]
        if t==b'+': return b.decode()
        if t==b'-': return Exception(b.decode())
        if t==b':': return int(b)
        if t==b'$':
            n=int(b)
            if n<0: return None
            d=s.f.read(n+2); return d[:-2]
        if t==b'*':
            n=int(b); return [s._read() for _ in range(n)]
        raise ValueError(l)
    def call(s,*args):
        s.s.sendall(s.enc(args)); return s._read()
    def pipe(s,cmds,batch=1000):
        out=[]
        for i in range(0,len(cmds),batch):
            chunk=cmds[i:i+batch]
            s.s.sendall(b"".join(s.enc(c) for c in chunk))
            for _ in chunk: out.append(s._read())
        return out

# X11b: a scan in the middle of normal traffic. Cache-aside reads with Zipf popularity (alpha 0.8) over 2,000,000 keys,
# cache sized for about 200,000 keys. After warm-up, a batch job reads 400,000 distinct one-off keys (scan:i, GET then SET),
# interleaved one-for-one with normal reads. Hit ratio of the NORMAL reads is reported per window of 200,000 normal reads.
import sys, time
import numpy as np
from resp import R
N = 2000000; ALPHA = 0.8; C = 200000; WARM = 3000000; BEFORE = 1000000; SCAN = 400000; AFTER = 2400000; W = 200000; B = 20
rng = np.random.default_rng(5)
p = np.arange(1, N + 1, dtype=np.float64) ** (-ALPHA); p /= p.sum(); cdf = np.cumsum(p)
normal = np.searchsorted(cdf, rng.random(WARM + BEFORE + SCAN + AFTER)).tolist()
r = R(); val = b"x" * 100
def used():
    for l in r.call("INFO", "memory").decode().splitlines():
        if l.startswith("used_memory:"): return int(l.split(":")[1])
def stream():
    i = 0
    for _ in range(WARM + BEFORE): yield ("n", normal[i]); i += 1
    for s in range(SCAN): yield ("n", normal[i]); i += 1; yield ("s", s)
    for _ in range(AFTER): yield ("n", normal[i]); i += 1
print(f"N={N}, alpha={ALPHA}, cache ~{C} keys, warm-up {WARM}, then windows of {W} normal reads; scan of {SCAN} one-off keys interleaved 1:1 during windows {BEFORE//W + 1}-{(BEFORE+SCAN)//W}")
for pol in sys.argv[1].split(","):
    r.call("CONFIG", "SET", "maxmemory", 0); r.call("FLUSHALL"); r.call("CONFIG", "SET", "maxmemory-policy", pol); r.call("CONFIG", "SET", "maxmemory-samples", 5)
    r.call("CONFIG", "SET", "maxmemory", used() + C * 160); r.call("CONFIG", "RESETSTAT")
    sock, f, enc = r.s, r.f, r.enc
    seen = 0; hits = 0; windows = []; t0 = time.time(); batch = []
    def flush(batch):
        global seen, hits
        sock.sendall(b"".join(enc(("GET", f"{'z' if k == 'n' else 'scan'}:{v}")) for k, v in batch))
        miss = []
        for k, v in batch:
            l = f.readline()
            if l[:2] == b"$-": miss.append((k, v))
            else:
                f.read(int(l[1:-2]) + 2)
            if k == "n":
                seen += 1
                if l[:2] != b"$-" and seen > WARM: hits += 1
                if seen > WARM and (seen - WARM) % W == 0: windows.append(hits / W); hits = 0
        if miss:
            ms = list(dict.fromkeys(miss))
            sock.sendall(b"".join(enc(("SET", f"{'z' if k == 'n' else 'scan'}:{v}", val)) for k, v in ms))
            for _ in ms: f.readline()
    for item in stream():
        batch.append(item)
        if len(batch) == B: flush(batch); batch = []
    if batch: flush(batch)
    dt = time.time() - t0
    print(f"{pol:14s} keys={r.call('DBSIZE')} req/s~{(WARM+BEFORE+2*SCAN+AFTER)/dt:.0f}  hit ratio per window: " + " ".join(f"{x:.3f}" for x in windows), flush=True)
r.call("CONFIG", "SET", "maxmemory", 0); r.call("CONFIG", "SET", "maxmemory-policy", "noeviction"); r.call("FLUSHALL")

#### script resp.py
import socket
class R:
    def __init__(s, port=7114):
        s.s = socket.create_connection(('127.0.0.1', port)); s.s.setsockopt(socket.IPPROTO_TCP, socket.TCP_NODELAY, 1); s.f = s.s.makefile('rb')
    @staticmethod
    def enc(args):
        o = [f"*{len(args)}\r\n".encode()]
        for a in args:
            if isinstance(a, str): a = a.encode()
            elif isinstance(a, (int, float)): a = str(a).encode()
            o.append(b"$%d\r\n" % len(a)); o.append(a); o.append(b"\r\n")
        return b"".join(o)
    def read(s):
        l = s.f.readline(); t, b = l[:1], l[1:-2]
        if t == b'+': return b.decode()
        if t == b'-': return Exception(b.decode())
        if t == b':': return int(b)
        if t == b'$':
            n = int(b)
            if n < 0: return None
            return s.f.read(n + 2)[:-2]
        if t == b'*': return [s.read() for _ in range(int(b))]
        raise ValueError(l)
    def call(s, *a): s.s.sendall(s.enc(a)); return s.read()
    def pipe(s, cmds, batch=1000):
        out = []
        for i in range(0, len(cmds), batch):
            ch = cmds[i:i + batch]; s.s.sendall(b"".join(s.enc(c) for c in ch)); out += [s.read() for _ in ch]
        return out
    def info(s, field, sec='stats'):
        for l in s.call('INFO', sec).decode().split('\r\n'):
            if l.startswith(field + ':'): return l.split(':', 1)[1]

#### script p1_paced.py
# P1: the X10 workload scaled down, run on Redis 8.10.3 at full speed and at a paced rate.
# Cache-aside reads (GET, SET on miss, 100-byte values) with Zipf popularity (exponent 0.8) over N keys,
# maxmemory set for about C keys (used_memory when empty + C * 160 bytes), maxmemory-samples 5.
# Full speed: FILL + WARM + MEAS requests pipelined 20 at a time, as X10 was.
# Paced: FILL requests at full speed to fill the cache, then requests sent one at a time at RATE a second,
# SETTLE seconds unmeasured, then MEAS_S seconds counted.
# usage: python3 p1_paced.py mode policy[:decay] ...   mode = fast | paced
import sys, time
import numpy as np
sys.path.insert(0, sys.argv[0].rsplit('/', 1)[0])
from resp import R
N, ALPHA, C = 800000, 0.8, 4000
FILL, WARM, MEAS = 200000, 600000, 400000
RATE, SETTLE, MEAS_S = 400, 600, 600
mode = sys.argv[1]
rng = np.random.default_rng(7)
p = np.arange(1, N + 1, dtype=np.float64) ** (-ALPHA); p /= p.sum(); cdf = np.cumsum(p)
total = FILL + (WARM + MEAS if mode == 'fast' else RATE * (SETTLE + MEAS_S) + 1000)
reqs = np.searchsorted(cdf, rng.random(total)).tolist()
r = R(); val = b'x' * 100
def used(): return int(r.info('used_memory', 'memory'))
def fast(seq, count_from):
    hits = 0; B = 20; sock, f, enc = r.s, r.f, r.enc
    for b0 in range(0, len(seq), B):
        ks = seq[b0:b0 + B]
        sock.sendall(b''.join(enc(('GET', f'z:{k}')) for k in ks)); miss = []
        for j, k in enumerate(ks):
            l = f.readline()
            if l[:2] == b'$-': miss.append(k)
            else:
                f.read(int(l[1:-2]) + 2)
                if b0 + j >= count_from: hits += 1
        if miss:
            ms = list(dict.fromkeys(miss)); sock.sendall(b''.join(enc(('SET', f'z:{k}', val)) for k in ms))
            for _ in ms: f.readline()
    return hits
print(f'mode={mode} N={N} alpha={ALPHA} C~{C} ' + (f'fill {FILL} + warm {WARM} + measured {MEAS} at full speed' if mode == 'fast' else f'fill {FILL} at full speed, then {RATE}/s ({RATE*60/C:.0f} reads a minute per cached key): {SETTLE} s unmeasured, {MEAS_S} s measured'), flush=True)
for spec in sys.argv[2:]:
    pol, _, decay = spec.partition(':'); decay = decay or '1'
    r.call('CONFIG', 'SET', 'maxmemory', 0); r.call('FLUSHALL'); r.call('CONFIG', 'SET', 'maxmemory-policy', pol)
    r.call('CONFIG', 'SET', 'maxmemory-samples', 5); r.call('CONFIG', 'SET', 'lfu-decay-time', decay); r.call('CONFIG', 'SET', 'lfu-log-factor', 10)
    r.call('CONFIG', 'SET', 'maxmemory', used() + C * 160); r.call('CONFIG', 'RESETSTAT')
    t0 = time.time()
    if mode == 'fast':
        hits = fast(reqs[:FILL + WARM + MEAS], FILL + WARM); n = MEAS
    else:
        fast(reqs[:FILL], 10**12)
        i = FILL; start = time.time(); hits = n = 0; lag = 0.0
        while True:
            el = time.time() - start
            if el >= SETTLE + MEAS_S: break
            k = reqs[i]; i += 1
            h = r.call('GET', f'z:{k}') is not None
            if not h: r.call('SET', f'z:{k}', val)
            if el >= SETTLE: n += 1; hits += h
            target = start + (i - FILL) / RATE
            d = target - time.time()
            if d > 0: time.sleep(d)
            else: lag = max(lag, -d)
    dt = time.time() - t0
    print(f'{pol:12s} lfu-decay-time {decay:>3s}  keys {r.call("DBSIZE"):5d}  hit ratio {hits/n:.4f} over {n} requests  evicted_keys {r.info("evicted_keys")}  wall {dt:.0f} s' + (f'  max lag {lag:.2f} s' if mode == 'paced' else f'  req/s {(FILL+WARM+MEAS)/dt:.0f}'), flush=True)
r.call('CONFIG', 'SET', 'maxmemory', 0); r.call('FLUSHALL')

#### script p1model.mjs
// Model mirror of P1 (p1_paced.py): Zipf 0.8 over 800,000 keys, capacity C keys.
// fast: FILL+WARM+MEAS requests at FASTRATE a second. paced: FILL at FASTRATE, then 400 a second for 600 s unmeasured + 600 s measured.
const base = '/Users/ryannel/Workspace/staycurrent/src/components/caching/';
const { SampledCache } = await import(base + 'expiration-and-eviction/sampled-cache.mjs');
const { zipf, seededRandom } = await import(base + 'shared/sim.mjs');
const [mode, C, FASTRATE] = [process.argv[2], +process.argv[3], +process.argv[4]];
const N = 800000, FILL = 200000, WARM = 600000, MEAS = 400000, RATE = 400, SETTLE = 600, MEAS_S = 600;
const z = zipf(N, 0.8);
for (const spec of process.argv.slice(5)) {
  const [pol, decay = '1'] = spec.split(':');
  const r = seededRandom(7), cr = seededRandom(99);
  const c = new SampledCache({ capacity: C, policy: pol, samples: 5, random: cr, decayMinutes: +decay });
  let hits = 0, n = 0, now = 0;
  const total = mode === 'fast' ? FILL + WARM + MEAS : FILL + RATE * (SETTLE + MEAS_S);
  for (let i = 0; i < total; i++) {
    now = i < FILL || mode === 'fast' ? i * 1000 / FASTRATE : FILL * 1000 / FASTRATE + (i - FILL) * 1000 / RATE;
    const k = z.sample(r); const h = c.get(k, now); if (!h) c.set(k, now);
    const counted = mode === 'fast' ? i >= FILL + WARM : i >= FILL + RATE * SETTLE;
    if (counted) { n++; hits += h; }
  }
  console.log(`model ${mode} C=${C} ${pol} decay ${decay}: hit ratio ${(hits / n).toFixed(4)} over ${n}`);

// M5: X10's workload (Zipf 0.8 over N keys, cache-aside) in the article's Redis-like model
// (sampled-cache.mjs), with requests spaced evenly at a chosen pace. Pace is given as reads a
// minute for each key the cache holds (rate = pace * C / 60). Runs warm up for max(WARM requests,
// WARMMIN simulated minutes), then count MEAS requests.
// usage: node pace.mjs N C pace policy[,policy...]   policy: random | lru | exact | lfu | lfu10 | lfu0
const base = '/Users/ryannel/Workspace/staycurrent/src/components/caching/';
const { SampledCache } = await import(base + 'expiration-and-eviction/sampled-cache.mjs');
const { zipf, seededRandom } = await import(base + 'shared/sim.mjs');
// An exact LRU: a doubly linked list over slots, most recent at the head.
class ExactLru { constructor(cap) { this.cap = cap; this.slot = new Map(); this.key = new Int32Array(cap); this.prev = new Int32Array(cap).fill(-1); this.next = new Int32Array(cap).fill(-1); this.head = -1; this.tail = -1; this.n = 0; }
  unlink(i) { const p = this.prev[i], n = this.next[i]; if (p >= 0) this.next[p] = n; else this.head = n; if (n >= 0) this.prev[n] = p; else this.tail = p; }
  front(i) { this.prev[i] = -1; this.next[i] = this.head; if (this.head >= 0) this.prev[this.head] = i; this.head = i; if (this.tail < 0) this.tail = i; }
  get(k) { const i = this.slot.get(k); if (i === undefined) return false; this.unlink(i); this.front(i); return true; }
  set(k) { let i; if (this.n < this.cap) i = this.n++; else { i = this.tail; this.unlink(i); this.slot.delete(this.key[i]); } this.key[i] = k; this.slot.set(k, i); this.front(i); } }
const [N, C, pace, pols] = [+process.argv[2], +process.argv[3], +process.argv[4], process.argv[5].split(',')];
const WARM = +(process.env.WARM ?? 3000000), MEAS = +(process.env.MEAS ?? 2000000), WARMMIN = +(process.env.WARMMIN ?? 30);
// RATE (requests a second) overrides pace; pace is then reported as derived from it.
const rate = process.env.RATE ? +process.env.RATE : pace * C / 60;
// The benchmark pace (240) reproduces X10 as run (3M + 2M requests, about a minute); slower paces also warm for WARMMIN simulated minutes.
const warm = (process.env.BENCH || (!process.env.RATE && pace >= 240)) ? WARM : Math.max(WARM, Math.ceil(WARMMIN * 60 * rate));
const z = zipf(N, 0.8);
for (const pol of pols) {
  const r = seededRandom(1), cr = seededRandom(99);
  let c;
  if (pol === 'exact') c = new ExactLru(C);
  else if (pol.startsWith('lfu')) c = new SampledCache({ capacity: C, policy: 'lfu', samples: 5, random: cr, decayMinutes: pol === 'lfu' ? 1 : +pol.slice(3) });
  else if (pol.startsWith('lru') && pol.length > 3) c = new SampledCache({ capacity: C, policy: 'lru', samples: +pol.slice(3), random: cr });
  else c = new SampledCache({ capacity: C, policy: pol, samples: 5, random: cr });
  let hits = 0; const t0 = Date.now();
  for (let i = 0; i < warm + MEAS; i++) {
    const k = z.sample(r), now = i * 1000 / rate;
    let h;
    h = c.get(k, now); if (!h) c.set(k, now);
    if (i >= warm && h) hits++;
  }
  console.log(JSON.stringify({ N, C, pace: +(rate * 60 / C).toFixed(3), rate: +rate.toFixed(2), policy: pol, warm, meas: MEAS, simMinutes: +((warm + MEAS) / rate / 60).toFixed(1), hit: +(hits / MEAS).toFixed(4), secs: (Date.now() - t0) / 1000 }));
}
