#### script resp.py
import socket
class R:
    def __init__(s,port=7104):
        s.s=socket.create_connection(("127.0.0.1",port)); s.s.setsockopt(socket.IPPROTO_TCP, socket.TCP_NODELAY,1); s.f=s.s.makefile('rb')
    @staticmethod
    def enc(args):
        o=[f"*{len(args)}\r\n".encode()]
        for a in args:
            if isinstance(a,str): a=a.encode()
            elif isinstance(a,int): a=str(a).encode()
            o.append(b"$%d\r\n"%len(a)); o.append(a); o.append(b"\r\n")
        return b"".join(o)
    def _read(s):
        l=s.f.readline()
        t=l[:1]; b=l[1:-2]
        if t==b'+': return b.decode()
        if t==b'-': return Exception(b.decode())
        if t==b':': return int(b)
        if t==b'$':
            n=int(b)
            if n<0: return None
            d=s.f.read(n+2); return d[:-2]
        if t==b'*':
            n=int(b); return [s._read() for _ in range(n)]
        raise ValueError(l)
    def call(s,*args):
        s.s.sendall(s.enc(args)); return s._read()
    def pipe(s,cmds,batch=1000):
        out=[]
        for i in range(0,len(cmds),batch):
            chunk=cmds[i:i+batch]
            s.s.sendall(b"".join(s.enc(c) for c in chunk))
            for _ in chunk: out.append(s._read())
        return out

# X11b: a scan in the middle of normal traffic. Cache-aside reads with Zipf popularity (alpha 0.8) over 2,000,000 keys,
# cache sized for about 200,000 keys. After warm-up, a batch job reads 400,000 distinct one-off keys (scan:i, GET then SET),
# interleaved one-for-one with normal reads. Hit ratio of the NORMAL reads is reported per window of 200,000 normal reads.
import sys, time
import numpy as np
from resp import R
N = 2000000; ALPHA = 0.8; C = 200000; WARM = 3000000; BEFORE = 1000000; SCAN = 400000; AFTER = 2400000; W = 200000; B = 20
rng = np.random.default_rng(5)
p = np.arange(1, N + 1, dtype=np.float64) ** (-ALPHA); p /= p.sum(); cdf = np.cumsum(p)
normal = np.searchsorted(cdf, rng.random(WARM + BEFORE + SCAN + AFTER)).tolist()
r = R(); val = b"x" * 100
def used():
    for l in r.call("INFO", "memory").decode().splitlines():
        if l.startswith("used_memory:"): return int(l.split(":")[1])
def stream():
    i = 0
    for _ in range(WARM + BEFORE): yield ("n", normal[i]); i += 1
    for s in range(SCAN): yield ("n", normal[i]); i += 1; yield ("s", s)
    for _ in range(AFTER): yield ("n", normal[i]); i += 1
print(f"N={N}, alpha={ALPHA}, cache ~{C} keys, warm-up {WARM}, then windows of {W} normal reads; scan of {SCAN} one-off keys interleaved 1:1 during windows {BEFORE//W + 1}-{(BEFORE+SCAN)//W}")
for pol in sys.argv[1].split(","):
    r.call("CONFIG", "SET", "maxmemory", 0); r.call("FLUSHALL"); r.call("CONFIG", "SET", "maxmemory-policy", pol); r.call("CONFIG", "SET", "maxmemory-samples", 5)
    r.call("CONFIG", "SET", "maxmemory", used() + C * 160); r.call("CONFIG", "RESETSTAT")
    sock, f, enc = r.s, r.f, r.enc
    seen = 0; hits = 0; windows = []; t0 = time.time(); batch = []
    def flush(batch):
        global seen, hits
        sock.sendall(b"".join(enc(("GET", f"{'z' if k == 'n' else 'scan'}:{v}")) for k, v in batch))
        miss = []
        for k, v in batch:
            l = f.readline()
            if l[:2] == b"$-": miss.append((k, v))
            else:
                f.read(int(l[1:-2]) + 2)
            if k == "n":
                seen += 1
                if l[:2] != b"$-" and seen > WARM: hits += 1
                if seen > WARM and (seen - WARM) % W == 0: windows.append(hits / W); hits = 0
        if miss:
            ms = list(dict.fromkeys(miss))
            sock.sendall(b"".join(enc(("SET", f"{'z' if k == 'n' else 'scan'}:{v}", val)) for k, v in ms))
            for _ in ms: f.readline()
    for item in stream():
        batch.append(item)
        if len(batch) == B: flush(batch); batch = []
    if batch: flush(batch)
    dt = time.time() - t0
    print(f"{pol:14s} keys={r.call('DBSIZE')} req/s~{(WARM+BEFORE+2*SCAN+AFTER)/dt:.0f}  hit ratio per window: " + " ".join(f"{x:.3f}" for x in windows), flush=True)
r.call("CONFIG", "SET", "maxmemory", 0); r.call("CONFIG", "SET", "maxmemory-policy", "noeviction"); r.call("FLUSHALL")
