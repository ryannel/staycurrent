## E7. Flushing a counter while it is being incremented: GETDEL against GET then DEL
## 200,000 INCRs from 20 clients (redis-benchmark), while a loop flushes the counter as fast as it can.
## Script:
# Usage: flush.sh atomic|split  — runs 200,000 INCRs from 20 clients while a flusher
# moves the counter into a running total every 10 ms, then compares totals.
MODE=$1
redis-cli -p 7102 DEL views:p > /dev/null
redis-benchmark -p 7102 -n 200000 -c 20 -q INCR views:p > bench.$MODE.out 2>&1 &
B=$!
total=0; flushes=0
while kill -0 $B 2>/dev/null; do
  if [ "$MODE" = atomic ]; then v=$(redis-cli -p 7102 GETDEL views:p)
  else v=$(redis-cli -p 7102 GET views:p); redis-cli -p 7102 DEL views:p > /dev/null; fi
  total=$((total + ${v:-0})); flushes=$((flushes+1))
done
v=$(redis-cli -p 7102 GETDEL views:p); total=$((total + ${v:-0}))
echo "mode=$MODE increments_sent=200000 flushed_total=$total flushes=$flushes lost=$((200000-total))"
## Output (two runs of each, interleaved):
mode=atomic increments_sent=200000 flushed_total=200000 flushes=594 lost=0
mode=split increments_sent=200000 flushed_total=100340 flushes=285 lost=99660
mode=atomic increments_sent=200000 flushed_total=200000 flushes=571 lost=0
mode=split increments_sent=200000 flushed_total=99958 flushes=284 lost=100042
## Note: in 'split' mode the gap between GET and DEL includes starting a second redis-cli process
## (a few milliseconds), so the loss is exaggerated against an application holding one connection.
## Any gap loses the increments that land inside it; GETDEL has no gap.
