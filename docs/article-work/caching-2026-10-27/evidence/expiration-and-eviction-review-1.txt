### resp7154.py
import socket
class R:
    def __init__(s, port=7154):
        s.s = socket.create_connection(('127.0.0.1', port)); s.f = s.s.makefile('rb')
    def cmd(s, *a):
        out = b'*%d\r\n' % len(a)
        for x in a:
            x = x if isinstance(x, bytes) else str(x).encode()
            out += b'$%d\r\n%s\r\n' % (len(x), x)
        s.s.sendall(out); return s.read()
    def read(s):
        l = s.f.readline(); t, rest = l[:1], l[1:-2]
        if t == b'+': return rest.decode()
        if t == b'-': return 'ERR:' + rest.decode()
        if t == b':': return int(rest)
        if t == b'$':
            n = int(rest)
            if n < 0: return None
            d = s.f.read(n + 2)[:-2]; return d
        if t == b'*':
            n = int(rest); return None if n < 0 else [s.read() for _ in range(n)]
        raise Exception(l)

### x10clock.mjs
// Separates sampling from the one-second clock for X10's smallest sizes, using the article's own
// Redis-like SampledCache model (uniform sampling + 16-entry pool). Requests are spaced at 80,000 a second
// (about what the writer's X10 runs achieved), so a 20,000-key cache turns over in ~0.3 s as on the real server.
const base = '/Users/ryannel/Workspace/staycurrent/src/components/caching/';
const { SampledCache } = await import(base + 'expiration-and-eviction/sampled-cache.mjs');
const { zipf, seededRandom } = await import(base + 'shared/sim.mjs');
const N = 4000000, WARM = 3000000, MEAS = 2000000, RATE = 80000;
const z = zipf(N, 0.8);
const [cap, ...cfgs] = process.argv.slice(2);
for (const cfg of cfgs) {
  const [policy, res, samples] = cfg.split(':');
  const r = seededRandom(1); const cr = seededRandom(99);
  const c = new SampledCache({ capacity: +cap, policy, samples: +samples, random: cr, clockResolution: +res });
  let hits = 0;
  for (let i = 0; i < WARM + MEAS; i++) {
    const k = z.sample(r); const now = i * 1000 / RATE;
    const h = c.get(k, now); if (!h) c.set(k, now);
    if (i >= WARM && h) hits++;
  }
  console.log(`cap=${cap} ${policy} clock=${res}ms samples=${samples}: hit ratio ${(hits / MEAS).toFixed(4)}`);
}

### scanclock.mjs
// Re-run Figure 9's scan model with the LRU clock at 1 s (as shipped), 1 ms, and an exact LRU,
// to see how much of the figure's LRU behaviour comes from the one-second clock at this compressed timescale.
const base = '/Users/ryannel/Workspace/staycurrent/src/components/caching/expiration-and-eviction/';
const { scanModel, meanBetween } = await import(base + 'scan-model.mjs');
const { SampledCache } = await import(base + 'sampled-cache.mjs');
const origClock = SampledCache.prototype.clock;
function run(res, samples, seed) {
  const C = class extends SampledCache {};
  // patch the class used by scanModel via prototype: set resolution per instance
  SampledCache.prototype.clock = function (now) { return Math.floor(now / (this.policy === 'lru' ? res : 1000)); };
  const sp = SampledCache.prototype.populate;
  const m = scanModel({}, seed);
  m.caches.lru.samples = samples;
  m.sim.advance(80e3);
  SampledCache.prototype.clock = origClock;
  return m;
}
for (const [label, res, samples] of [['1 s clock, 5 samples (shipped)', 1000, 5], ['1 ms clock, 5 samples', 1, 5], ['1 ms clock, 64 samples (~exact LRU)', 1, 64], ['1 s clock, 64 samples', 1000, 64]]) {
  const rows = [];
  for (const seed of [1, 2, 3, 4, 5]) {
    const m = run(res, samples, seed);
    rows.push([meanBetween(m, 'lru', 5, 20), meanBetween(m, 'lru', 20, 30), meanBetween(m, 'lru', 30, 35), meanBetween(m, 'lru', 35, 60), meanBetween(m, 'lfu', 5, 20), meanBetween(m, 'lfu', 20, 30)]);
  }
  const avg = rows[0].map((_, i) => rows.reduce((s, r) => s + r[i], 0) / rows.length);
  console.log(label.padEnd(38), 'LRU before', avg[0].toFixed(3), 'during', avg[1].toFixed(3), `(${((1 - avg[1] / avg[0]) * 100).toFixed(0)}% drop)`, 'just after', avg[2].toFixed(3), 'later', avg[3].toFixed(3), '| LFU before', avg[4].toFixed(3), 'during', avg[5].toFixed(3));
}
