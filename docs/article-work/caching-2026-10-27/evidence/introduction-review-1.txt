## Script used
# Independent reproduction: Zipf(s=1) over 100,000 keys through Redis on port 7151 with a memory limit.
# Compares allkeys-lru (samples 5 and 64), allkeys-random, an exact-LRU simulation, and a paced run.
import socket, random, bisect, time, sys
from collections import OrderedDict
N=100000; S=1.0
def conn():
    s=socket.create_connection(("127.0.0.1",7151)); return s, s.makefile('rb')
sock,f=conn()
def enc(*a):
    out=[b"*%d\r\n"%len(a)]
    for x in a:
        x=x if isinstance(x,bytes) else str(x).encode(); out.append(b"$%d\r\n%s\r\n"%(len(x),x))
    return b"".join(out)
def read():
    l=f.readline()
    t=l[:1]; r=l[1:-2]
    if t in (b'+',b'-'): return r
    if t==b':': return int(r)
    if t==b'$':
        n=int(r)
        if n<0: return None
        d=f.read(n+2); return d[:-2]
    if t==b'*': return [read() for _ in range(int(r))]
def cmd(*a): sock.sendall(enc(*a)); return read()
def pipe(cmds):
    sock.sendall(b"".join(enc(*c) for c in cmds)); return [read() for _ in cmds]
def info(field):
    for line in cmd("INFO","memory").decode().split("\r\n"):
        if line.startswith(field+":"): return int(line.split(":")[1])
w=[1/k**S for k in range(1,N+1)]; cum=[];t=0
for x in w: t+=x; cum.append(t)
rng=random.Random(7)
def draw(): return bisect.bisect_left(cum, rng.random()*t)
VAL=b"x"*100
def setup(C,policy,samples):
    cmd("FLUSHALL"); cmd("CONFIG","SET","maxmemory","0")
    base=info("used_memory")
    pipe([("SET","item:%d"%k,VAL) for k in range(C)])
    full=info("used_memory")
    cmd("CONFIG","SET","maxmemory-policy",policy); cmd("CONFIG","SET","maxmemory-samples",str(samples))
    cmd("CONFIG","SET","maxmemory",str(full))
    return full
def run(seq,batch=100,pace=None):
    h=m=0
    for i in range(0,len(seq),batch):
        ks=seq[i:i+batch]
        res=pipe([("GET","item:%d"%k) for k in ks])
        miss=[k for k,r in zip(ks,res) if r is None]
        h+=len(ks)-len(miss); m+=len(miss)
        if miss: pipe([("SET","item:%d"%k,VAL) for k in miss])
        if pace: time.sleep(pace)
    return h,m
def exact_lru(seq,C,start):
    c=OrderedDict(); h=n=0
    for i,k in enumerate(seq):
        hit=k in c
        if hit: c.move_to_end(k)
        else:
            c[k]=1
            if len(c)>C: c.popitem(last=False)
        if i>=start: n+=1; h+=hit
    return h/n
mode=sys.argv[1]
if mode=="fast":
    seq=[draw() for _ in range(600000)]; warm=200000
    for C in (1000,10000):
        print(f"exact-LRU sim C={C}: {exact_lru(seq,C,warm)*100:.1f}%",flush=True)
        for policy,samples in (("allkeys-lru",5),("allkeys-lru",64),("allkeys-random",5)):
            mm=setup(C,policy,samples); t0=time.time(); run(seq[:warm]); h,m=run(seq[warm:]); dt=time.time()-t0
            print(f"C={C} {policy} samples={samples} maxmemory={mm} held={cmd('DBSIZE')} hit={h/(h+m)*100:.1f}% over {h+m} requests, {len(seq)/dt:,.0f} req/s",flush=True)
elif mode=="paced":
    # Slow enough that the cache turns over in well over a second: ~2,000 requests a second, C=1000.
    seq=[draw() for _ in range(150000)]; warm=50000
    C=1000
    print(f"exact-LRU sim C={C}: {exact_lru(seq,C,warm)*100:.1f}%",flush=True)
    mm=setup(C,"allkeys-lru",5); t0=time.time(); run(seq[:warm],batch=10,pace=0.012); h,m=run(seq[warm:],batch=10,pace=0.012); dt=time.time()-t0
    print(f"PACED C={C} allkeys-lru samples=5 maxmemory={mm} held={cmd('DBSIZE')} hit={h/(h+m)*100:.1f}% over {h+m} requests, {len(seq)/dt:,.0f} req/s",flush=True)
