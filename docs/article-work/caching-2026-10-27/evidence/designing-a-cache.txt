# E1: bytes per entry for the design's key families, from used_memory deltas.
import json, random, redis
r = redis.Redis(port=7112)
random.seed(12)
def product(i):
    doc = {"id": i, "name": f"Trail running shoe model {i}", "brand": "Northfield",
           "price_pence": 4900 + i % 3000, "currency": "GBP",
           "description": ("Light shoe for rough ground with a grippy sole and a padded collar. " * 22)[:1500],
           "images": [f"https://img.example/p/{i}/{k}.jpg" for k in range(4)],
           "category": i % 2000, "rating": 4.4, "reviews": 128}
    return json.dumps(doc, separators=(",", ":"))
def measure(label, n, write):
    r.flushall(); r.config_set("maxmemory", 0)
    before = r.info("memory")["used_memory"]
    p = r.pipeline(transaction=False)
    for i in range(n):
        write(p, i)
        if i % 500 == 499: p.execute()
    p.execute()
    after = r.info("memory")["used_memory"]
    print(f"{label}: {n} keys, used_memory +{after-before} bytes, {(after-before)/n:.0f} bytes a key, dbsize {r.dbsize()}")
sizes = [len(product(i)) for i in range(1000)]
print("product JSON length: mean", sum(sizes)/len(sizes), "min", min(sizes), "max", max(sizes))
measure("product doc, SET EX 86400 with 10% jitter", 20000,
        lambda p, i: p.set(f"shop:product:v3:{i}", product(i), ex=int(86400 * random.uniform(0.9, 1.1))))
measure("stock, SET EX 86400 with 10% jitter", 20000,
        lambda p, i: p.set(f"shop:stock:v1:{i}", str(random.randint(0, 400)), ex=int(86400 * random.uniform(0.9, 1.1))))
trend = json.dumps([{"id": i, "name": f"Trail running shoe model {i}", "price_pence": 4900, "image": f"https://img.example/p/{i}/0.jpg"} for i in range(20)], separators=(",", ":"))
print("trending value length", len(trend))
r.flushall()
r.set("shop:trending:v1", trend, ex=600)
print("MEMORY USAGE shop:trending:v1 =", r.memory_usage("shop:trending:v1"))
r.flushall()

================ E2: allkeys-lru and allkeys-lfu hit ratios, paced Zipf 0.9 cache-aside ================
N = 100,000 products, 200-byte values, maxmemory set for about 30,000 keys, 5,000 reads/s paced,
120 s warm-up then 120 s measured. 'ideal' = share of reads to the top-k products for the k held.
Che approximation for the same k (budget-model.mjs lruShare): 0.7731 for LRU.
--- script: e2_lru.py
# E2: Redis allkeys-lru hit ratio against Che's approximation, Zipf exponent 0.9,
# 100,000 items, 200-byte values, cache-aside client (GET, then SET on a miss),
# paced at RATE requests a second in pipelined batches of 50 every 10 ms.
import bisect, math, random, sys, time, redis
N, S = 100_000, 0.9
RATE = int(sys.argv[1]) if len(sys.argv) > 1 else 5000
WARM, MEASURE = int(sys.argv[2]) if len(sys.argv) > 2 else 120, int(sys.argv[3]) if len(sys.argv) > 3 else 120
POLICY = sys.argv[4] if len(sys.argv) > 4 else "allkeys-lru"
r = redis.Redis(port=7112)
w = [1 / (i + 1) ** S for i in range(N)]; H = sum(w)
cum = []; c = 0
for x in w: c += x / H; cum.append(c)
rnd = random.Random(9)
val = b"x" * 200
r.flushall(); r.config_set("maxmemory", 0); r.config_set("maxmemory-policy", POLICY)
base = r.info("memory")["used_memory"]
p = r.pipeline(transaction=False)
for i in range(20000):
    p.set(f"shop:product:v3:{i}", val)
    if i % 1000 == 999: p.execute()
p.execute()
per = (r.info("memory")["used_memory"] - base) / 20000
r.flushall()
target = 30000
r.config_set("maxmemory", int(base + target * per))
print(f"policy {POLICY}, {per:.0f} bytes a key, maxmemory {base + target * per:.0f}, rate {RATE}/s, warm {WARM}s, measure {MEASURE}s")
def run(seconds):
    batch = 50; interval = batch / RATE
    start = time.time(); n = 0
    while time.time() - start < seconds:
        keys = [f"shop:product:v3:{bisect.bisect_left(cum, rnd.random())}" for _ in range(batch)]
        vals = r.mget(keys)  # one round trip; counts a hit or miss per key
        miss = [k for k, v in zip(keys, vals) if v is None]
        if miss:
            pp = r.pipeline(transaction=False)
            for k in miss: pp.set(k, val)
            pp.execute()
        n += batch
        sleep = start + n / RATE - time.time()
        if sleep > 0: time.sleep(sleep)
    return n, time.time() - start
run(WARM)
s0 = r.info("stats")
n, el = run(MEASURE)
s1 = r.info("stats")
h = s1["keyspace_hits"] - s0["keyspace_hits"]; m = s1["keyspace_misses"] - s0["keyspace_misses"]
held = r.dbsize()
top = cum[held - 1]
print(f"requests {n} in {el:.1f}s ({n/el:.0f}/s); keys held {held}; hit ratio {h/(h+m):.4f}; ideal (top {held}) {top:.4f}; evicted_keys {s1['evicted_keys']}")

#!/bin/bash
# E3: the design's commands, end to end, on Redis 8.10.3 (port 7112).
c() { echo "\$ redis-cli $*"; redis-cli -p 7112 "$@"; }
ca() { u=$1; shift; echo "\$ redis-cli --user $u $*"; redis-cli -p 7112 --user $u --pass "$u-test" --no-auth-warning "$@"; }
redis-cli -p 7112 FLUSHALL >/dev/null
echo "# Configuration for the product cache"
c CONFIG SET maxmemory 3500mb maxmemory-policy allkeys-lru save "" appendonly no
c CONFIG GET maxmemory-policy
c CONFIG GET maxmemory-samples
echo "# Users: the application, and the relay that sends deletes from the outbox"
c ACL SETUSER shop-app on '>shop-app-test' resetkeys '~shop:*' resetchannels -@all +@read +@write +@connection -@dangerous
c ACL SETUSER shop-relay on '>shop-relay-test' resetkeys '~shop:product:*' '~shop:stock:*' resetchannels -@all +del +unlink +ping
echo "# Read path: one round trip for details and stock"
ca shop-app MGET shop:product:v3:42 shop:stock:v1:42
echo "# Miss on both: take a lease on each, in one round trip"
ca shop-app SET shop:product:v3:42 lease:r1 NX PX 2000
ca shop-app SET shop:stock:v1:42 lease:r1 NX PX 2000
echo "# A price change commits; the relay deletes the product key before the fill lands"
ca shop-relay DEL shop:product:v3:42
echo "# The reader's fills: the product fill is refused because its lease is gone; the stock fill lands"
ca shop-app SET shop:product:v3:42 '{"id":42,"price_pence":3900}' IFEQ lease:r1 EX 87120
ca shop-app SET shop:stock:v1:42 17 IFEQ lease:r1 EX 81400

ca shop-app MGET shop:product:v3:42 shop:stock:v1:42

ca shop-relay DEL shop:stock:v1:42

ca shop-app SET shop:trending:v1 '{"soft_until":1793260800,"items":[42,7,19]}' EX 600
echo "# Soft-expired: one request takes the rebuild lock, the others keep serving the old value"
ca shop-app SET shop:lock:trending:v1 tok-a NX PX 5000
ca shop-app SET shop:lock:trending:v1 tok-b NX PX 5000
ca shop-app SET shop:trending:v1 '{"soft_until":1793260860,"items":[7,42,19]}' EX 600
ca shop-app DELEX shop:lock:trending:v1 IFEQ tok-a
echo "# Not found: a short-lived marker, deleted when the product is created"
ca shop-app SET shop:product:v3:999999 __NOTFOUND__ EX 30
echo "# What the application may not do"
ca shop-app FLUSHALL
ca shop-app KEYS 'shop:*'
ca shop-app CONFIG SET maxmemory 0
ca shop-app GET session:abc

ca shop-relay GET shop:product:v3:42

c ACL DELUSER shop-app shop-relay
