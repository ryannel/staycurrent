import sys
# usage: gen.py kind n valuelen
kind, n, vl = sys.argv[1], int(sys.argv[2]), int(sys.argv[3])
out = sys.stdout.buffer
def cmd(*a):
    b = b"*%d\r\n" % len(a)
    for x in a:
        x = x if isinstance(x, bytes) else str(x).encode()
        b += b"$%d\r\n%s\r\n" % (len(x), x)
    return b
val = b"v" * vl
for i in range(n):
    if kind == "str":
        out.write(cmd("SET", "user:%08d" % i, val))
    elif kind == "strttl":
        out.write(cmd("SET", "user:%08d" % i, val, "EX", 3600))
    elif kind == "hash5":
        out.write(cmd("HSET", "user:%08d" % i, "a", val, "b", val, "c", val, "d", val, "e", val))
    elif kind == "fields":   # n fields in one hash
        out.write(cmd("HSET", "bighash", "f%08d" % i, val))
    elif kind == "setint":
        out.write(cmd("SADD", "bigset", i))
    elif kind == "setstr":
        out.write(cmd("SADD", "bigset", "m%08d" % i))
    elif kind == "zset":
        out.write(cmd("ZADD", "bigz", i, "m%08d" % i))

import socket,time
class C:
    def __init__(s,name="c",port=7204):
        s.name=name; s.s=socket.create_connection(("127.0.0.1",port)); s.f=s.s.makefile("rb")
    def send(s,*a):
        b=b"*%d\r\n"%len(a)
        for x in a:
            x=x if isinstance(x,bytes) else str(x).encode()
            b+=b"$%d\r\n%s\r\n"%(len(x),x)
        s.s.sendall(b)
    def read(s):
        line=s.f.readline(); t=line[:1]
        if t==b"+": return line[1:-2].decode()
        if t==b"-": return "ERR:"+line[1:-2].decode()
        if t==b":": return int(line[1:-2])
        if t==b"$":
            n=int(line[1:-2]); return None if n<0 else s.f.read(n+2)[:-2].decode()
        if t==b"*":
            n=int(line[1:-2]); return None if n<0 else [s.read() for _ in range(n)]
        if t==b"_": s.f.readline() if False else None; return None
        raise Exception(line)
    def do(s,*a,show=True):
        s.send(*a); r=s.read()
        if show: print("%s> %s\n   %r"%(s.name," ".join(map(str,a))[:150],r))
        return r
